第 2 周：Gin + REST API 基础 —— 总结与验收

五天里你把一个 main.go 手工作坊，升级成了有目录规范、统一响应、中间件流水线和文档的像样的后端项目。今天做三件事：串知识地图、按参考骨架冲刺主线项目、逐条过验收标准。验收不过周不算完。

本周知识地图

Day主题核心产出一句话复盘自测动作
D1Gin 路由gin.New/Default、c.Param/c.Query、r.Group、NoRoute路由声明式匹配替代字符串比较，参数从 Context 取盲写 /users/:id 路由 + curl 验证
D2绑定与验证ShouldBindJSON、binding tag、pkg/response.go结构体即 schema：解析、校验、统一 {code,message,data} 一步到位故意发非法 JSON，拿到 400+1004
D3中间件Logger/Recovery、Timer、手写 CORS、鉴权雏形洋葱模型：c.Next() 分进出，c.Abort() 熔断内层画出 A→B→handler→B→A 执行图
D4项目分层cmd/server + internal 三层 + map/RWMutex 仓库依赖单向 handler→service→repo→model，下层不知上层口述 import cycle 成因与解法
D5接口测试curl 五件套、Postman collection、README 文档状态码 + 业务码双验收；错误路径（400/404/500）也是功能Runner 一键全绿 + 陌生人照 README 跑通

贯穿线：「API 设计约定」（手册第十三节）本周落地——统一响应 {code,message,data}、HTTP 状态码管大类 + 业务码 1001–1005 管细分、成功一律 200/201+code 0。下周接数据库时这套外壳不再改动。

3. 参考代码骨架

pkg/response/response.gopackage response

import ("net/http"; "github.com/gin-gonic/gin")

type Body struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
    Data    any    `json:"data"`
}

const (
    CodeOK = 0; CodeUserNotFound = 1001; CodeBadPassword = 1002
    CodeForbidden = 1003; CodeInvalidParam = 1004; CodeTokenInvalid = 1005
)

func OK(c *gin.Context, data any)      { c.JSON(http.StatusOK, Body{CodeOK, "success", data}) }
func Created(c *gin.Context, data any) { c.JSON(http.StatusCreated, Body{CodeOK, "success", data}) }
func Fail(c *gin.Context, st, code int, msg string) { c.JSON(st, Body{code, msg, nil}) }
internal/repository/user_repo.go（节选）var ErrUserNotFound = errors.New("用户不存在")

type UserRepo struct {
    mu     sync.RWMutex
    nextID int
    users  map[int]model.User
}

func NewUserRepo() *UserRepo { return &UserRepo{nextID: 1, users: make(map[int]model.User)} }

func (r *UserRepo) Get(id int) (model.User, error) {
    r.mu.RLock()
    defer r.mu.RUnlock()
    u, ok := r.users[id]
    if !ok { return model.User{}, ErrUserNotFound }
    return u, nil
}
// Create/List/Update/Delete 同模式：写路径 Lock，读路径 RLock
internal/service/user_service.go（节选）type UserService struct { repo *repository.UserRepo }

func NewUserService(repo *repository.UserRepo) *UserService { return &UserService{repo: repo} }

func (s *UserService) Create(dto model.CreateUserDTO) (model.User, error) {
    for _, u := range s.repo.List() {
        if u.Username == dto.Username {
            return model.User{}, fmt.Errorf("用户名 %q 已被占用", dto.Username)
        }
    }
    role := dto.Role
    if role == "" { role = "user" }
    return s.repo.Create(model.User{Username: dto.Username, Email: dto.Email,
        Password: dto.Password, Role: role, CreatedAt: time.Now()}), nil
}

func (s *UserService) Get(id int) (model.User, error) { return s.repo.Get(id) }
func IsNotFound(err error) bool { return errors.Is(err, repository.ErrUserNotFound) }
internal/handler/user_handler.go（节选）type UserHandler struct { svc *service.UserService }

func (h *UserHandler) Register(r *gin.RouterGroup) {
    r.GET("/users", h.List)
    r.POST("/users", h.Create)          // ShouldBindJSON → svc.Create → Created
    r.GET("/users/:id", h.Get)
    r.PUT("/users/:id", h.Update)
    r.DELETE("/users/:id", h.Delete)
}

func (h *UserHandler) Get(c *gin.Context) {
    id, err := strconv.Atoi(c.Param("id"))
    if err != nil {
        response.Fail(c, http.StatusBadRequest, response.CodeInvalidParam, "id 必须是数字")
        return
    }
    u, err := h.svc.Get(id)
    if service.IsNotFound(err) {
        response.Fail(c, http.StatusNotFound, response.CodeUserNotFound, "用户不存在")
        return
    }
    if err != nil {
        response.Fail(c, http.StatusInternalServerError, 5000, "服务繁忙")
        return
    }
    response.OK(c, u)
}
// List/Create/Update/Delete 按 day4 讲义补全
cmd/server/main.gofunc main() {
    userRepo := repository.NewUserRepo()
    articleRepo := repository.NewArticleRepo()
    userSvc := service.NewUserService(userRepo)
    articleSvc := service.NewArticleService(articleRepo, userSvc)

    r := gin.New()
    r.Use(gin.Logger(), gin.Recovery(), middleware.CORS(), middleware.Timer())
    r.NoRoute(func(c *gin.Context) {
        response.Fail(c, http.StatusNotFound, 404, "接口不存在: "+c.Request.URL.Path)
    })

    api := r.Group("/api/v1")
    handler.NewUserHandler(userSvc).Register(api)
    handler.NewArticleHandler(articleSvc).Register(api)

    r.Run(":8080")
}

4. curl 验收命令（逐条执行，状态码与 body 双确认）

shell# —— 用户 CRUD 全链路 ——
$ curl -si -X POST localhost:8080/api/v1/users -H "Content-Type: application/json" \
  -d '{"username":"ada","email":"ada@example.com","password":"passw0rd-123"}'
# 期望: HTTP 201 · {"code":0,"message":"success","data":{"id":1,...}}

$ curl -si localhost:8080/api/v1/users | head -n 1          # 期望 200
$ curl -si localhost:8080/api/v1/users/1 | head -n 1        # 期望 200
$ curl -si -X PUT localhost:8080/api/v1/users/1 -H "Content-Type: application/json" \
  -d '{"username":"ada_lovelace","email":"ada@example.com","password":"passw0rd-123"}'
# 期望: 200 + code 0，username 已更新
$ curl -si -X DELETE localhost:8080/api/v1/users/1 | head -n 1   # 期望 200
$ curl -si localhost:8080/api/v1/users/1 | head -n 1             # 期望 404！再查即不存在

# —— 错误路径 ——
$ curl -si -X POST localhost:8080/api/v1/users -H "Content-Type: application/json" \
  -d '{"username":"ab","email":"x","password":"1"}'
# 期望: 400 + {"code":1004,"message":"Username 长度不足 3；...","data":null}

# —— 文章 ——
$ curl -si -X POST localhost:8080/api/v1/articles -H "Content-Type: application/json" \
  -d '{"title":"Go 真香","content":"...","author_id":99}'
# 期望: 404 + code 1001（作者不存在）；author_id 换成真实 id 后期望 201
$ curl -s "localhost:8080/api/v1/articles?page=1&size=10"   # 期望 200 code 0

# —— CORS 预检 ——
$ curl -si -X OPTIONS localhost:8080/api/v1/users \
  -H "Origin: http://localhost:5173" -H "Access-Control-Request-Method: POST"
# 期望: 204 + Access-Control-Allow-Origin/Methods/Headers 三头齐全

# —— Recovery 兜底（保留 /debug/boom 或临时加一条）——
$ curl -si localhost:8080/debug/boom | head -n 1            # 期望 500
$ curl -s localhost:8080/api/v1/users -o /dev/null -w "%{http_code}\n"  # 期望 200：服务没死

验收标准（全部通过才算毕业）

自评三卡

代码量体检三层每层都写过 ≥ 100 行？handler 里还残留业务 if 判断吗？service 里还能搜到 gin.Context 吗？——有则回 day4 返工。
卡点体检本周最疼的坑是哪个：小写字段绑定失败？CORS 预检？import cycle？把「现象 → 原因 → 解法」三行写进 README 的 FAQ 小节。
速度体检从现在起，从零搭出带统一响应 + 中间件的空项目骨架，能否在 30 分钟内完成？第 3 周要接 MySQL，手速不达标周末会被挤爆。

常见卡点（历届学员周报高频词）

卡点典型症状解法
三层拆完不会组装main.go 里 new 顺序混乱、字段全 nil 的 panic严格按 repo→service→handler 自底向上 new；每个 NewXxx 构造函数字段赋值齐全再返回
响应双写日志 superfluous WriteHeader，body 出现两个 JSON 拼接全局搜索 response.Fail 与 c.JSON，确认每分支后紧跟 return
required 误伤 0/falseauthor_id:0 或分页 size 为 0 被拒，行为诡异允许零值的字段改 omitempty,gte=0，缺省逻辑放 service
JSON 输出 {}repository 明明存了数据，接口却空对象检查字段是否大写开头、json tag 是否写、是否把未导出结构体直接扔进了 gin.H
并发下 map 崩Runner 一跑并发就 fatal error: concurrent map writes锁必须覆盖读和写；读用 RLock；确认 defer Unlock 在 Lock 的同一函数内
OPTIONS 被鉴权拦死预检返回 401，浏览器永远 CORS 失败CORS 中间件注册在 Auth 之前，且对 OPTIONS AbortWithStatus(204)（见 day3）

下周预告：为什么要真数据库，Docker 登场

内存 map 的三宗罪，本周你一定已经撞上：

  • 重启即灭：Ctrl+C 再 go run，用户全没了——第 1 周「重启待办仍在」的验收标准本周被我们自己打破了，第 3 周用 MySQL 把它捡回来。
  • 单进程独食：内存数据没法多实例共享，也撑不了任何数据量，List 全表遍历纯属暴力。
  • 没有查询语言：day4 里手写 for 循环找重名用户，第 3 周变成一句 WHERE username = ?，外加索引提速。

第 3 周（MySQL + GORM + Docker 入门）：用 docker compose up 三十秒起一个带数据卷的 MySQL，GORM AutoMigrate 建表，把本周项目原样改造——handler/service 几乎不动，只重写 repository 层。你会亲眼看到分层架构的回报。周末记得做一件事：把 docker-compose.yml 起 MySQL 的官方教程扫一眼目录，混个眼熟即可，周一开讲。
