知识讲义

1. slice：带容量的动态数组

Go 的数组 [3]int 把长度写死在类型里，几乎不用；日常说的"数组"是 slice——指向底层数组的视图，含起始指针、len（当前长度）、cap（可增长上限）三样东西。

slice.gopackage main

import fmt

func main() {
    var s []int            // nil slice：len=0 cap=0，可以直接 append
    for i := 1; i <= 5; i++ {
        s = append(s, i*10) // append 返回新 slice，必须接回变量
        fmt.Printf("第 %d 次: %v len=%d cap=%d\n", i, s, len(s), cap(s))
    }
    // 输出（Go 1.22+ 的翻倍策略）：
    // [10] len=1 cap=1 / [10 20] len=2 cap=2 / [10 20 30] len=3 cap=4 ← 扩容：换新数组并拷贝旧数据
    // [10 20 30 40] len=4 cap=4 / [10 20 30 40 50] len=5 cap=8 ← 再扩容

    list := make([]string, 0, 100) // 已知数量就预分配容量，全程不再扩容
    list = append(list, "学 Go")

    nums := make([]int, 3)         // make([]T, n) 得到 n 个零值，可下标赋值
    nums[0], nums[1] = 5, 6
    // nums[3] = 7 → panic: index out of range（Go 不会自动长大）
    fmt.Println(list, nums, cap(nums))
}

JS 的 arr.push(x) 原地修改并返回新长度；Go 的 append 返回新 slice（扩容后甚至指向另一个数组）。写成裸一行 append(s, x) 等于什么都没加——本周最高频 bug。

2. slice 的三个坑：视图共享、copy、删除元素

slice2.gopackage main

import fmt

func main() {
    a := []int{1, 2, 3, 4, 5}
    b := a[1:3]  // [2 3]：不是拷贝！和 a 共用底层数组
    b[0] = 99
    fmt.Println(a)  // [1 99 3 4 5] ← a 被改了

    c := make([]int, len(b))
    copy(c, b)     // 真拷贝，之后互不影响（也可用 slices.Clone）
    c[0] = 7
    fmt.Println(b, c) // [99 3] [7 3]

    // 更阴的坑：append 到"还有容量"的子切片，会踩掉后面的元素
    d := a[:2]        // len=2 cap=5
    d = append(d, 100) // 直接写进 a[2] 的位置
    fmt.Println(a, d)  // [1 99 100 4 5] [1 99 100]

    // 删除下标 i 的元素（Go 没有 splice，标准写法是两段拼接）
    items := []string{"a", "b", "c", "d"}
    i := 2
    items = append(items[:i], items[i+1:]...) // ... 展开可变参数；注意它复用底层数组
    fmt.Println(items) // [a b d]
}

3. map：键值存储

map.gopackage main

import fmt

type Todo struct {
    ID    int
    Title string
    Done  bool
}

func main() {
    byID := make(map[int]Todo)                       // make 创建
    scores := map[string]int{"语文": 92, "数学": 85} // 字面量创建

    byID[1] = Todo{ID: 1, Title: "买菜"}   // 增 / 改是同一语法
    byID[2] = Todo{ID: 2, Title: "写 Go 作业"}
    delete(byID, 1)                          // 删：键不存在也不报错

    fmt.Println("查不到的直接取值:", byID[1]) // 返回零值 Todo，掩盖了"没查到"

    // 推荐：二返回值 ok 区分"零值"与"不存在"
    if t, ok := byID[2]; ok {
        fmt.Println("存在:", t.Title)
    } else {
        fmt.Println("不存在")
    }

    fmt.Println("长度:", len(byID))
    for k, n := range scores { fmt.Println(k, n) } // 遍历顺序随机，别依赖

    // map 的 value 是 struct 时不能直接改字段（不可寻址）
    byID[3] = Todo{ID: 3, Title: "学 map"}
    // byID[3].Done = true   // 编译失败：cannot assign to struct field in map
    t := byID[3]
    t.Done = true
    byID[3] = t                // 取出→改→塞回
    // 更省事：一开始就存指针 map[int]*Todo，直接 byID[3].Done = true
    fmt.Println(byID[3])
}

只声明不初始化的 var m map[string]int 是 nil map：读可以，写直接 panic assignment to entry in nil map。函数返回 map 时也别忘初始化。

JS 里 obj[key] || 默认值 满天飞；Go 读不到返回零值而不是 undefined。而且 for k := range m 顺序被运行时故意打乱——想"按创建顺序输出待办"，就用 slice 存顺序、map 只做 ID 索引。

4. defer：把收尾工作交给语言

defer 把一次调用推迟到当前函数返回时执行；多个 defer 按后进先出（LIFO）运行，参数在 defer 语句那一刻就已求值。

defer.gopackage main

import (
    fmt
    os
)

func orderDemo() {
    defer fmt.Println("3 最后执行")
    defer fmt.Println("2 倒数第二")
    x := 10
    defer fmt.Println("1 参数即时求值:", x) // 打印 10，不是 11
    x++
    fmt.Println("0 主体")
}
// 输出：0 主体 / 1 参数即时求值: 10 / 2 倒数第二 / 3 最后执行

func readFile(path string) (string, error) {
    f, err := os.Open(path)
    if err != nil {
        return "", err // defer 还没注册，直接返回
    }
    defer f.Close() // ← 标准姿势：拿到资源，下一行就 defer 释放

    buf := make([]byte, 128)
    n, err := f.Read(buf)
    if err != nil && n == 0 {
        return "", err // 这条 return 同样会执行 f.Close()
    }
    return string(buf[:n]), nil
}

func main() {
    orderDemo()
    text, err := readFile("todos.json") // 首次运行文件还不存在，会走 err 分支
    fmt.Println(text, err)
}

规则一句话：成对资源（open/close、加锁/解锁、开始/结束）在拿到的下一行就写 defer。但 defer 是函数级的——循环里 defer 1000 次会攒 1000 个句柄到函数结束才释放，需要每轮释放就把逻辑抽成小函数。

5. 文件读写：os.ReadFile / os.WriteFile

file.gopackage main

import (
    fmt
    os
    strings
)

func main() {
    // 写：0o644 = 属主可读写、其他人只读（Go 八进制字面量用 0o 前缀）
    if err := os.WriteFile("note.txt", []byte("第一行\n第二行\n"), 0o644); err != nil {
        fmt.Println("写入失败:", err)
        return
    }

    // 读：一次拿到全部字节
    data, err := os.ReadFile("note.txt")
    if os.IsNotExist(err) {
        fmt.Println("文件还不存在") // 首次启动是正常情况，不算错误
        return
    } if err != nil {
        fmt.Println("读取失败:", err)
        return
    }
    for i, line := range strings.Split(strings.TrimSpace(string(data)), "\n") {
        fmt.Printf("%d: %s\n", i+1, line)
    }
    fmt.Println("共", len(data), "字节")
}

os.WriteFile 是整体覆盖而不是追加。待办数据量很小，每次全量覆盖一个 JSON 文件最简单可靠；第 3 周接上 MySQL 后就不自己管文件了。

需要追加日志那样写时，用 f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644) 拿到 *os.File，立刻 defer f.Close()，再 fmt.Fprintln(f, "...")——defer 和文件 I/O 两个知识点在这里合流。

6. encoding/json + 落盘：周末项目的持久层原型

Go 用字段名（或 struct tag）在 JSON 键与字段之间做映射，Marshal 序列化、Unmarshal 反序列化：

todo_store.gopackage main

import (
    encoding/json
    fmt
    os
)

type Todo struct {
    ID        int    `json:"id"`
    Title     string `json:"title"`
    Done      bool   `json:"done"`
    CreatedAt string `json:"created_at,omitempty"` // 零值时省略该键
}

const storeFile = "todos.json"

// LoadTodos：文件不存在 / 内容为空都返回空列表，只有真错误才报错
func LoadTodos(path string) ([]Todo, error) {
    data, err := os.ReadFile(path)
    if os.IsNotExist(err) { return []Todo{}, nil }
    if err != nil { return nil, fmt.Errorf("读取 %s: %w", path, err) }
    if len(data) == 0 { return []Todo{}, nil }

    todos := []Todo{}
    if err := json.Unmarshal(data, &todos); err != nil {
        return nil, fmt.Errorf("解析 JSON: %w", err) // %w 包装错误，Day 4 细讲
    }
    return todos, nil
}

// SaveTodos：MarshalIndent 生成带缩进的可读 JSON
func SaveTodos(path string, todos []Todo) error {
    data, err := json.MarshalIndent(todos, "", "  ")
    if err != nil { return fmt.Errorf("序列化: %w", err) }
    return os.WriteFile(path, data, 0o644)
}

func NextID(todos []Todo) int {
    max := 0
    for _, t := range todos { if t.ID > max { max = t.ID } }
    return max + 1
}

func main() {
    todos, err := LoadTodos(storeFile)
    if err != nil { fmt.Println(err); return }
    fmt.Printf("启动时读到 %d 条待办\n", len(todos))

    // 反序列化一个对象：对应 Day 5 里解析 POST 请求体
    var in Todo
    err = json.Unmarshal([]byte(`{"id":7,"title":"从请求体来"}`), &in)
    if err != nil { fmt.Println("请求体非法 JSON:", err) } // Day 5 映射为 400
    fmt.Printf("解析结果: %+v\n", in) // %+v 连字段名一起打印

    todos = append(todos, Todo{ID: NextID(todos), Title: "学习 JSON 持久化", CreatedAt: "2026-09-23"})
    if err := SaveTodos(storeFile, todos); err != nil {
        fmt.Println("保存失败:", err)
        return
    }
    fmt.Println("已写入", storeFile, "共", len(todos), "条")
}

对照 JS：① 字段必须大写开头才会被导出，忘写 tag 就得到 {"ID":1} 而不是 {"id":1}；② Unmarshal 要传指针 &todos（它要往调用方的变量里写数据）；③ JSON 中多余的键被忽略、缺失的键保持零值，不像 JSON.parse 那样全量还原。

三个边界：nil slice 序列化成 null 而非 []（前端直接 .map 会炸，所以初始化写 []T{}）；JSON 数字默认转 float64，超大 ID 需 json.Number；map 的键只能是字符串类类型。

今日练习

练习 1：观测 append 扩容（20 分钟）

循环 append 20 个元素，每次打印 len/cap，记录扩容发生在第几次；再用 make([]int, 0, 20) 重写一遍确认全程不扩容。把结论写进笔记：为什么"已知数量就预分配"能省内存拷贝。

answer.gofunc growLab() {
    var s []int
    for i := 1; i <= 20; i++ {
        old := cap(s)
        s = append(s, i)
        if cap(s) != old {
            fmt.Printf("第 %2d 次 append 触发扩容: cap %d -> %d\n", i, old, cap(s))
        }
    }
    // 典型输出：cap 1 -> 2 -> 4 -> 8 -> 16 -> 32
    pre := make([]int, 0, 20)
    for i := 1; i <= 20; i++ { pre = append(pre, i) }
    fmt.Println("预分配 cap=", cap(pre), "len=", len(pre)) // 20 20，零次扩容
}
练习 2：词频统计 + 有序输出（30 分钟）

统计一段英文里每个词的出现次数（map），再按"次数从多到少"打印。因为 map 遍历无序，必须先把键取进 slice 再排序。提示：strings.Fields + sort.Slice。

answer.gofunc wordFreq() {
    text := "go is fast and go is simple and i love go"
    counts := map[string]int{}
    for _, w := range strings.Fields(text) {
        counts[w]++ // 键不存在时零值 0 直接 +1，不需要先判断
    }
    words := make([]string, 0, len(counts))
    for w := range counts { words = append(words, w) }
    sort.Slice(words, func(i, j int) bool {
        if counts[words[i]] != counts[words[j]] { return counts[words[i]] > counts[words[j]] } // 次数降序
        return words[i] < words[j] // 次数相同按字典序，结果才稳定
    })
    for _, w := range words { fmt.Printf("%-6s %d\n", w, counts[w]) }
}

需要 import ("fmt"; "sort"; "strings")。多跑几次确认输出顺序一致，然后删掉排序那段再看它变乱——这个对比能帮你彻底记住 map 无序。

练习 3：给待办存储加查找，验证"重启不丢数据"（35 分钟）

手敲跑通讲义 6 的 todo_store.go，再补一个 FindByID。步骤：连续 go run . 三次 → 看条数 1→2→3 → cat todos.json 肉眼看 JSON → 把文件内容改成非法 JSON 再跑一次，读一读 LoadTodos 的报错文本。

answer.gofunc FindByID(todos []Todo, id int) (Todo, bool) {
    for _, t := range todos { if t.ID == id { return t, true } }
    return Todo{}, false // 零值 + false；Day 5 里这个 false 就是 404
}
练习 4：每日 CLI 待办 add / list / done（40 分钟）

用 os.Args 读参数，复用练习 3 的存储函数做小 CLI（规划里说明它只是每日练习，不是周末项目）：go run . add 学习Go / go run . list / go run . done 2。done 找不到 ID 时打印 未找到待办 #N 并用 os.Exit(1) 结束。注意：改元素要用 todos[i].Done = true（for _, t := range 里的 t 是拷贝）。这题的 list/add/done 逻辑，就是 Day 5 三个 handler 的前身。

answer.gofunc main() {
    args := os.Args[1:] // os.Args[0] 是程序名，参数从 1 开始
    todos, err := LoadTodos(storeFile)
    if err != nil { fmt.Println(err); os.Exit(1) }

    switch { // 无标签 switch：case 从上往下第一个为真者执行，故 len==0 先兜底
    case len(args) == 0:
        fmt.Println("用法: todo add <标题> | list | done <id>")
    case args[0] == "list":
        for _, t := range todos {
            box := " "
            if t.Done { box = "x" }
            fmt.Printf("  [%s] #%d %s\n", box, t.ID, t.Title)
        }
    case args[0] == "add" && len(args) > 1:
        todos = append(todos, Todo{ID: NextID(todos), Title: args[1]})
        fmt.Printf("已新增 #%d %s\n", todos[len(todos)-1].ID, args[1])
        err = SaveTodos(storeFile, todos)
    case args[0] == "done" && len(args) > 1:
        id, e := strconv.Atoi(args[1]) // 字符串转 int，自带解析错误
        if e != nil { fmt.Println("id 必须是数字"); os.Exit(1) }
        found := false
        for i := range todos { // 用下标改元素本体；for _, t 里的 t 是拷贝
            if todos[i].ID == id { todos[i].Done = true; found = true; break }
        }
        if !found { fmt.Printf("未找到待办 #%d\n", id); os.Exit(1) }
        fmt.Printf("已完成 #%d\n", id)
        err = SaveTodos(storeFile, todos)
    }
    if err != nil { fmt.Println("保存失败:", err); os.Exit(1) }
}

需要 import ("fmt"; "os"; "strconv")，并把讲义 6 的 Todo / LoadTodos / SaveTodos / NextID / storeFile 放在同一个包里。
