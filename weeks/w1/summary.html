store.go（Day 3 的持久化，项目里唯一真正碰磁盘的地方）package main

import (
    "encoding/json"
    "fmt"
    "os"
)

const storeFile = "todos.json"     // 路径只写这一次，main.go 里用 NewFileRepo(storeFile)

func LoadTodos(path string) ([]Todo, error) {
    data, err := os.ReadFile(path)
    if os.IsNotExist(err) { return []Todo{}, nil }  // 首次运行没文件：空列表，不是错误
    if err != nil { return nil, fmt.Errorf("读取 %s: %w", path, err) }
    if len(data) == 0 { return []Todo{}, nil }         // 空文件会让 Unmarshal 报错，先兜住
    todos := []Todo{}                                // 初始化成 []，JSON 才会输出 [] 而非 null
    if err := json.Unmarshal(data, &todos); err != nil { return nil, fmt.Errorf("解析 JSON: %w", err) }
    return todos, nil
}
func SaveTodos(path string, todos []Todo) error {
    data, err := json.MarshalIndent(todos, "", "  ")  // 缩进 2 空格，方便人肉 cat 检查
    if err != nil { return fmt.Errorf("序列化: %w", err) }
    return os.WriteFile(path, data, 0o644)            // 权限不对 → error → handler 兜成 500
}

3. 每个 handler 的参考骨架

main.go（组装：全部路由挂在一个 struct 上）type server struct { repo TodoRepository } // 只依赖接口（Day 4）

func main() {
    repo, err := NewFileRepo(storeFile)   // storeFile = "todos.json"，在 store.go 里；启动即 LoadTodos
    if err != nil { log.Fatal("启动失败:", err) }
    s := &server{repo: repo}

    mux := http.NewServeMux()
    mux.HandleFunc("GET /health", s.health)
    mux.HandleFunc("GET /todos", s.listTodos)
    mux.HandleFunc("POST /todos", s.createTodo)
    mux.HandleFunc("GET /todos/{id}", s.getTodo)
    mux.HandleFunc("PATCH /todos/{id}", s.completeTodo)
    mux.HandleFunc("DELETE /todos/{id}", s.deleteTodo)
    srv := &http.Server{Addr: ":8080", Handler: mux, ReadHeaderTimeout: 5 * time.Second}
    // Day 5 讲义把 Handler 换成了包一层的 logged（方法 + 路径 + 耗时）；这里传 mux 也能跑
    log.Fatal(srv.ListenAndServe())          // 返回值必须检查，端口被占要立刻知道
}
main.go（两个公共工具：一个出口 + 一个错误翻译）func writeJSON(w http.ResponseWriter, status int, v any) {
    w.Header().Set("Content-Type", "application/json; charset=utf-8") // 必须在 WriteHeader 之前
    w.WriteHeader(status)
    if err := json.NewEncoder(w).Encode(v); err != nil { log.Println("encode:", err) }
}

func (s *server) fail(w http.ResponseWriter, err error) {  // 本周状态码的全部逻辑
    switch {
    case errors.Is(err, ErrEmptyTitle), errors.Is(err, ErrBadID), errors.Is(err, ErrBadJSON):
        writeJSON(w, http.StatusBadRequest, map[string]string{"error": err.Error()})
    case errors.Is(err, ErrNotFound):
        writeJSON(w, http.StatusNotFound, map[string]string{"error": "待办不存在"})
    default:
        log.Println("internal:", err)       // 详情进日志，客户端只看到一句人话
        writeJSON(w, http.StatusInternalServerError, map[string]string{"error": "服务内部错误"})
    }
}
main.go（POST：读 body → 校验 → 落盘 → 201）func (s *server) createTodo(w http.ResponseWriter, r *http.Request) {
    var in struct { Title string `json:"title"` }
    if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
        s.fail(w, fmt.Errorf("decode: %w", ErrBadJSON)); return // 每个分支写完响应必须 return
    }
    t, err := s.repo.Create(in.Title)       // FileRepo 内部：改内存 + SaveTodos 落盘
    if err != nil { s.fail(w, err); return } // 空 title → 400，写盘失败 → 500
    w.Header().Set("Location", "/todos/"+strconv.Itoa(t.ID)) // Itoa：int → string（Atoi 的反向）
    writeJSON(w, http.StatusCreated, t)     // 201，不是 200
}
main.go（PATCH / DELETE：PathValue 与 204）func (s *server) completeTodo(w http.ResponseWriter, r *http.Request) {
    id, err := strconv.Atoi(r.PathValue("id"))
    if err != nil { s.fail(w, ErrBadID); return }
    t, err := s.repo.Complete(id)
    if err != nil { s.fail(w, err); return }
    writeJSON(w, http.StatusOK, t)
}
func (s *server) deleteTodo(w http.ResponseWriter, r *http.Request) {
    id, err := strconv.Atoi(r.PathValue("id"))
    if err != nil { s.fail(w, ErrBadID); return }
    if err := s.repo.Delete(id); err != nil { s.fail(w, err); return }
    w.WriteHeader(http.StatusNoContent)     // 204：成功但没有 body
}
repo.go（Day 4 的地基：接口 + 内存实现，handler 只认它；FileRepo 在 filerepo.go）type TodoRepository interface {
    List() ([]Todo, error)
    Get(id int) (Todo, error)
    Create(title string) (Todo, error)
    Complete(id int) (Todo, error)
    Delete(id int) error
}
var _ TodoRepository = (*MemRepo)(nil)   // 编译期契约检查：漏方法就编译失败
var _ TodoRepository = (*FileRepo)(nil)

// 五个方法的第一行都是这两句（Day 4 讲义 6 的锁口径；第 5 周 day1–day3 才正式教 goroutine 与锁，今天照抄位置即可）：
//   r.mu.Lock(); defer r.mu.Unlock()
func (r *MemRepo) List() ([]Todo, error) {
    r.mu.Lock()
    defer r.mu.Unlock()
    return append([]Todo{}, r.items...), nil  // 拷一份出去，别让外面改到内部切片
}
filerepo.go（FileRepo 的写操作：改内存 → SaveTodos 落盘 → 返回错误）func (f *FileRepo) Create(title string) (Todo, error) {
    t, err := f.inner.Create(title)
    if err != nil { return Todo{}, err }   // ErrEmptyTitle 直接上抛，handler 翻译成 400
    items, err := f.inner.List()
    if err != nil { return t, err }
    return t, SaveTodos(f.path, items)      // os.WriteFile 失败 → 兜底分支 500
}

$ curl -si -X PATCH localhost:8080/todos/1 | head -n 1                    # 200

$ curl -si -X DELETE localhost:8080/todos/1 | head -n 1                   # 204

$ chmod 000 todos.json                                                    # 制造 500：写盘失败
$ curl -si -X POST localhost:8080/todos -d '{"title":"500"}' | head -n 1  # 期望 500，服务日志有原因
$ chmod 644 todos.json

# —— 重启不丢数据 ——
$ # Ctrl+C 停掉服务，再 go run .，然后：
$ curl -s localhost:8080/todos               # 期望：列表还在（启动时 LoadTodos 回来了）
$ curl -s -o /dev/null -w "%{http_code}\n" localhost:8080/nothing # 404（未注册路径）

卡住时按这个顺序自查，别看别的：① 端口通不通（curl -v 有没有连上）→ ② 路由 pattern 有没有写错（方法拼错就是 404/405）→ ③ handler 有没有 return（两个 JSON 拼一起就是没 return）→ ④ SaveTodos 有没有被调用（在里面加一行 log.Println）→ ⑤ JSON 字段是不是大写开头。

下周预告：Gin 是什么，为什么要换掉裸 net/http

这五天你已经亲手体会了标准库的"好"和"烦"。烦的地方正是框架存在的理由：

  • 重复样板：每个 handler 都要 Decode + 判错 + writeJSON；参数校验（必填、长度、范围）全靠手写 if。Gin 的 ShouldBindJSON + binding tag 一行搞定解析与校验。
  • 响应不统一：/nothing 的 404 是纯文本，方法不对是 405 空 body，前端拿到会懵。Gin 可以 NoRoute 统一兜底，配合 pkg/response 输出结构化 {code,message,data}。
  • 中间件要手写：Day 5 那个包一层的 logged 就是雏形，写多了全是胶水。Gin 提供洋葱模型的 c.Next()/c.Abort()，日志、恢复 panic、CORS、鉴权串成流水线。
  • 分组与文档：/api/v1 前缀、按资源分组、路由清单导出成 Postman collection，标准库都得自己造。

Gin 不是替代 net/http，而是包在它外面：r.Run(":8080") 里面仍然是一个 http.Server，gin.Context 里塞的就是你熟悉的 ResponseWriter 和 *Request。所以 Day 5 学到的东西一分都不浪费，反而成为你理解框架的地图。第 2 周项目：内存版用户 + 文章 CRUD API，验收标准是"Postman 完成完整增删改查"——本周的 TodoRepository 思路会直接升级成分层的 handler → service → repository。

周末记得做一件事：把 Gin 官方 Gin 文档 的 Routing / Binding / Middleware 三节只扫目录，知道有这些东西即可，周一开讲。另外把本周项目 push 到 GitHub（README 写清启动方式和接口表）——第 7 周部署时它会成为你的第一个作品。
