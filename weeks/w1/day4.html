知识讲义

1. interface：隐式实现

接口只声明"有哪些方法"，谁实现了它不需要任何声明——方法齐了就自动满足，这叫隐式实现。

iface.gopackage main

import "fmt"

// 接口：只约定能力，不关心谁具备
type Shape interface {
    Area() float64
    Perimeter() float64
}

type Rect struct { W, H float64 }
type Circle struct { R float64 }

func (r Rect) Area() float64      { return r.W * r.H }
func (r Rect) Perimeter() float64 { return 2 * (r.W + r.H) }
func (c Circle) Area() float64      { return 3.14159 * c.R * c.R }
func (c Circle) Perimeter() float64 { return 2 * 3.14159 * c.R }

// 参数写接口：Rect / Circle 都能传，因为方法齐了（没有 implements 关键字）
func describe(s Shape) { fmt.Printf("%T 面积=%.2f 周长=%.2f\n", s, s.Area(), s.Perimeter()) }

func main() {
    shapes := []Shape{Rect{W: 3, H: 4}, Circle{R: 2}} // 接口切片：一个 slice 装两种类型
    for _, one := range shapes { describe(one) }
    var s Shape = Rect{W: 1, H: 1}
    fmt.Printf("接口里的具体类型: %T\n", s) // 接口值同时存着 (动态类型, 值)
    // 删掉 Rect 的 Area → 编译期报错：Rect does not implement Shape (missing method Area)
}

main.Circle 面积=12.57 周长=12.57
接口里的具体类型: main.Rect

JS 是"跑起来才发现 obj.quack is not a function"；TypeScript 是结构化类型（形状对就算实现，但仍要标注/断言）；Go 是隐式实现 + 编译期检查。所以 Go 的接口通常由使用方定义（"我需要能 Area 的东西"），而不是实现方宣布"我实现了 Shape"。惯例：单方法接口最有用（io.Reader、fmt.Stringer、明天的 http.Handler），别给每个 struct 都配一个接口。

2. 类型断言、type switch 与空接口

assert.go（片段，接 §1 的 package main）func assertDemo() {
    var s Shape = Circle{R: 2}
    // 断言：带 ok 的二返回值形式永远安全（不带 ok 且类型不符会 panic）
    if c, ok := s.(Circle); ok { fmt.Println("是圆，半径", c.R) }
    if _, ok := s.(Rect); !ok { fmt.Println("不是矩形") }
    // type switch：按具体类型分支
    switch v := s.(type) {
    case Circle: fmt.Println("圆，v.R =", v.R)
    case Rect:   fmt.Println("矩形，v.W =", v.W)
    case nil:    fmt.Println("接口本身是 nil")
    default:     fmt.Println("未知类型")
    }
    // 空接口 any（= interface{}）：什么都能装，但也丢失全部信息
    bag := map[string]any{"n": 1, "s": "go", "b": true}
    for k, val := range bag { fmt.Printf("%s -> %T\n", k, val) }
}

两个坑：① 本页底部「常见坑」表里 「到处用 any」那一行说的就是这里——参数能用具体类型就用，只有真的"什么都能装"（未知 JSON 结构、通用容器）才用 any；② 把"类型为 *T、值为 nil"的指针赋给 error 接口，接口本身不是 nil，于是 if err != nil 永远成立。无错时老老实实 return nil。

3. error 接口、哨兵错误与自定义错误类型

error 只是内置接口 type error interface { Error() string }，任何实现 Error() string 的类型都能当错误返回——这就是"错误就是值"。§2–§4 都是片段（只给类型和函数，没有 package/import/main），两种跑法任选：① 每段自己套上 Day 2 那个起手模板（package main + import ("errors"; "fmt"; "strings") + func main(){ assertDemo() }），并把 §1 的 Shape / Rect / Circle 和下面这三行的 Todo 复制进去；② 直接抄 §4 末尾那份合并完整版，一次跑完 §2+§3+§4。

    Title     string `json:"title"`
    Done      bool   `json:"done"`

err.go（片段：需要 errors / fmt / strings 三个 import 和上面的 Todo）// ① 哨兵错误：包级变量，调用方用 errors.Is 比对
var (
    ErrNotFound   = errors.New("todo not found")
    ErrEmptyTitle = errors.New("title is required")
)

// ② 自定义错误类型：需要携带上下文（哪个字段、收到什么值）时用它
type ValidationError struct { Field, Got string }
func (e *ValidationError) Error() string { return fmt.Sprintf("field %s invalid, got %q", e.Field, e.Got) }

func CreateTodo(title string) (Todo, error) {
    title = strings.TrimSpace(title)
    if title == "" {
        // %w 把具体类型的错误挂进链条：errors.As / errors.Is 都还能取到
        return Todo{}, fmt.Errorf("create: %w", &ValidationError{Field: "title", Got: title})
    }
    return Todo{ID: 1, Title: title}, nil
}

func errDemo() {
    _, err := CreateTodo("   ")
    fmt.Println("错误文本:", err) // create: field title invalid, got ""
    var ve *ValidationError
    fmt.Println("As 取回类型:", errors.As(err, &ve), "字段:", ve.Field) // true title
    fmt.Println("Is 匹配哨兵:", errors.Is(err, ErrEmptyTitle)) // false：链上挂的不是它
}

Go 里 err.Error() 才是"报错文本"，对应 JS 的 e.message；Go 没有 throw/try/catch，也没有"未捕获异常一路冒泡"。要么当场处理，要么 return err 上抛，编译器还盯着你别把 err 接了不用。

顺手记两个动词：① %q = 把字符串带双引号打印（fmt.Sprintf("got %q", "") 得到 got ""），空串、含空格、含换行的值一眼就能看出边界；相关的 %+v 是"连字段名一起打印"（Day 3 §6 用过），%T 是打印类型。② 上面这段里 errors.As(err, &ve) 的意思是"顺着错误链找某个类型"，找到了就把值写进 ve 并返回 true —— 和 errors.Is 找"同一个变量"是一对。

4. 错误包装：%w、errors.Is、errors.As

后端调用是分层的（handler → service → repository）。底层只知道"没找到"，上层需要给用户一句人话。Go 的做法是逐层包装，用 %w 把原错误挂在链上。

wrap.go（片段：三层调用，需要 §3 的 ErrNotFound / ErrEmptyTitle 在同一目录）func readFromDisk(path string) ([]byte, error) {
    if path == "" { return nil, fmt.Errorf("open %q: %w", path, ErrNotFound) }
    return []byte("[]"), nil
}
func loadTodos(path string) (string, error) {
    data, err := readFromDisk(path)
    // 再包一层上下文：%w 保留链条（可判断），%v 只是拼文本（会断链）
    if err != nil { return "", fmt.Errorf("load todos: %w", err) }
    return string(data), nil
}
func handlerDemo() {
    _, err := loadTodos("")
    fmt.Println("完整错误链:", err) // load todos: open "": todo not found
    // Is 沿链判断"到底是哪个哨兵"——明天的 HTTP 状态码就靠这段决定
    switch {
    case errors.Is(err, ErrNotFound):   fmt.Println("--> 404 Not Found")
    case errors.Is(err, ErrEmptyTitle): fmt.Println("--> 400 Bad Request")
    default:                            fmt.Println("--> 500 Internal Server Error")
    }
    fmt.Println("== 比较:", err == ErrNotFound) // false ← 用 == 判断错误是错误写法
}

三条纪律：① 只在真正能处理的地方处理，否则 return fmt.Errorf("上下文: %w", err) 上抛；② 包装文本用小写、无标点，最外层再拼成给用户的话；③ 判断一律 errors.Is（哨兵）/errors.As（类型），绝不比较 err.Error() == "…"（改文案就全崩）。

    "errors"
    "fmt"
    "strings"

type Todo struct {
    ID        int    `json:"id"`
    Title     string `json:"title"`
    Done      bool   `json:"done"`

}

    ErrNotFound   = errors.New("todo not found")

func CreateTodo(title string) (Todo, error) {
    title = strings.TrimSpace(title)
    if title == "" {
        return Todo{}, fmt.Errorf("create: %w", &ValidationError{Field: "title", Got: title})
    }
    return Todo{ID: 1, Title: title}, nil
}

    if path == "" { return nil, fmt.Errorf("open %q: %w", path, ErrNotFound) }
    return []byte("[]"), nil
}

    data, err := readFromDisk(path)
    if err != nil { return "", fmt.Errorf("load todos: %w", err) }
    return string(data), nil

    fmt.Println("错误文本:", err)
    var ve *ValidationError
    fmt.Println("As 取回类型:", errors.As(err, &ve), "字段:", ve.Field)
    fmt.Println("Is 匹配哨兵:", errors.Is(err, ErrEmptyTitle))

    // —— §4：三层包装 + 沿链判断 ——
    _, err = loadTodos("")
    fmt.Println("完整错误链:", err)
    switch {
    case errors.Is(err, ErrNotFound):   fmt.Println("--> 404 Not Found")
    case errors.Is(err, ErrEmptyTitle): fmt.Println("--> 400 Bad Request")
    default:                            fmt.Println("--> 500 Internal Server Error")
    }
    fmt.Println("== 比较:", err == ErrNotFound) // false：包装过之后 == 就失效了

    // —— §2：类型断言（这里用 Todo 的 id 当"值"演示，不依赖 §1 的 Shape）——
    var bag any = Todo{ID: 7, Title: "断言"}
    if t, ok := bag.(Todo); ok { fmt.Println("断言成功:", t.ID, t.Title) }
    if _, ok := bag.(string); !ok { fmt.Println("不是 string（带 ok 的断言不会 panic）") }
    switch v := bag.(type) {
    case Todo:   fmt.Println("type switch 命中 Todo:", v.Title)
    case nil:    fmt.Println("any 里是 nil")
    default:     fmt.Println("其他类型")
    }
}

第 2 周预览 · internal/todo/todo.go（今天不用敲）package todo // 包名：小写、单个单词、不用复数

import (
    "encoding/json"
    "errors"
    "fmt"
    "os"
)

// 大写开头 = 导出（跨包可见）；小写 = 包内私有。Go 没有 public/private 关键字
type Todo struct {
    ID    int    `json:"id"`
    Title string `json:"title"`
    Done  bool   `json:"done"`
}

var ErrNotFound = errors.New("todo not found") // 导出的哨兵错误，调用方 errors.Is 判断

func Load(path string) ([]Todo, error) {
    b, err := os.ReadFile(path)
    if os.IsNotExist(err) { return []Todo{}, nil } // 第一次运行没文件，不算错误
    if err != nil { return nil, fmt.Errorf("load: %w", err) }
    var todos []Todo
    if err := json.Unmarshal(b, &todos); err != nil { return nil, fmt.Errorf("decode: %w", err) }
    return todos, nil
}

func Save(path string, todos []Todo) error { // 就是 Day 3 的 SaveTodos，只是搬进包里
    b, err := json.MarshalIndent(todos, "", "  ")
    if err != nil { return fmt.Errorf("encode: %w", err) }
    return os.WriteFile(path, b, 0o644)
}

规则说明
package main vs 库包main 包编译成可执行文件且必须有 func main()；其他包是库，只能被 import，用 go build/go test 验证
import 路径是"模块路径 + 目录"，没有 ./todo、../util 这种相对写法；别名 import t "…/internal/todo" 合法，点号导入别用
未使用的 import编译错误（不是 warning）。临时占位可写 _ "fmt"，日常用 goimports 自动整理
循环 importA 引 B、B 引 A 直接编译失败。解法：公共类型抽到第三个包，或在使用方定义接口
internal/ 目录目录名有魔法——只有同一父项目里的代码能 import 它，用来隔开"内部实现"和"给别人用的库"；第 2 周正式用它组织 model / repository / handler

gofmt 和 goimports 是什么：gofmt -w . 只做排版（缩进用 Tab、对齐、去多余空行），标准库自带；goimports = gofmt + 自动补上你用到的 import、删掉没用的，所以要单独装：go install golang.org/x/tools/cmd/goimports@latest，然后把编辑器的"保存时格式化"指到它（VS Code 的 Go 插件默认就是这么配的）。JS 同学把它理解成"Prettier + 自动 import"就行。有了它，"未使用的 import 编译不过"这条规则基本不会烦到你。

go.mod 里那行 go 1.22 是语言版本下限：写低于 1.22，明天 mux.HandleFunc("GET /todos/{id}", …) 的方法路由就不生效。不确定就跑 go version 看看本地版本（本周内容需要 Go 1.22 及以上）。

对比 npm：Go 没有"装进 node_modules"这一步，import 语句本身就是依赖声明，go mod tidy 负责下载并写进 go.mod/go.sum；版本采用"最小版本选择"，也不会出现同一依赖 20 份拷贝。

6. 实战：待办仓储接口 + 内存实现

明天 HTTP handler 的直接依赖。接口在使用方（main 包）定义，实现可以是内存、JSON 文件、以后的 MySQL——换实现不动 handler。下面三个文件全放进 todo-api/、全部 package main，最后一条 go run . 跑起来（func main() 只在 main.go 里出现一次）。

todo-api/model.gopackage main

import "errors"

// 本周唯一的模型定义：Day 3 的存储、Day 5 的 handler 都用这一个 Todo
type Todo struct {
    ID        int    `json:"id"`
    Title     string `json:"title"`
    Done      bool   `json:"done"`
    CreatedAt string `json:"created_at,omitempty"` // 不赋值就整个键都不出现（Day 3 的老朋友）
}

var (
    ErrNotFound   = errors.New("todo not found")    // 明天 -> 404
    ErrEmptyTitle = errors.New("title is required") // 明天 -> 400
)

锁是什么（三行讲完，第 5 周 day1–day3 才正式教 goroutine 与锁）：sync.Mutex 就是试衣间的门闩——Lock() 进去反锁，别人来 Lock() 只能排队等；Unlock() 出来才放行。为什么要锁：明天的 HTTP 服务会同时处理很多请求，两次 Create 挤在一起可能拿到同一个 nextID，数据就互相盖掉了。今天你只要照抄"进函数第一行 r.mu.Lock()、第二行 defer r.mu.Unlock()"这个位置即可，别自己发明锁。

todo-api/repo.gopackage main

import (
    "fmt"
    "strings"
    "sync"
)

// 仓储接口：只约定能力，不关心数据存在哪里
// 每个读方法都带 error：内存实现返回 nil，文件实现就能返回真实的读写错误
type TodoRepository interface {
    List() ([]Todo, error)
    Get(id int) (Todo, error)
    Create(title string) (Todo, error)
    Complete(id int) (Todo, error)
    Delete(id int) error
}

// 内存实现。mutex 应对明天 HTTP 并发；含 mutex 的类型只能有指针方法
type MemRepo struct {
    mu     sync.Mutex
    items  []Todo
    nextID int
}

func NewMemRepo() *MemRepo { return &MemRepo{nextID: 1} }

func (r *MemRepo) indexOf(id int) int { // 小写方法 = 包内私有辅助函数
    for i := range r.items { if r.items[i].ID == id { return i } }
    return -1
}

func (r *MemRepo) List() ([]Todo, error) {
    r.mu.Lock(); defer r.mu.Unlock()
    return append([]Todo{}, r.items...), nil // 返回副本，外部改不到内部数据；内存版 error 恒为 nil
}
func (r *MemRepo) Create(title string) (Todo, error) {
    title = strings.TrimSpace(title)
    if title == "" { return Todo{}, ErrEmptyTitle }
    r.mu.Lock(); defer r.mu.Unlock()
    t := Todo{ID: r.nextID, Title: title}
    r.items, r.nextID = append(r.items, t), r.nextID+1
    return t, nil
}
func (r *MemRepo) Get(id int) (Todo, error) {
    r.mu.Lock(); defer r.mu.Unlock()
    i := r.indexOf(id)             // 先取下标再判：if i := ...; i < 0 {} 里的 i 只活在 if 语句里
    if i < 0 { return Todo{}, fmt.Errorf("get %d: %w", id, ErrNotFound) }
    return r.items[i], nil
}
func (r *MemRepo) Complete(id int) (Todo, error) {
    r.mu.Lock(); defer r.mu.Unlock()
    i := r.indexOf(id)
    if i < 0 { return Todo{}, fmt.Errorf("complete %d: %w", id, ErrNotFound) }
    r.items[i].Done = true // 下标访问，改的是元素本体（Day 2 的结论）
    return r.items[i], nil
}
func (r *MemRepo) Delete(id int) error {
    r.mu.Lock(); defer r.mu.Unlock()
    i := r.indexOf(id)
    if i < 0 { return fmt.Errorf("delete %d: %w", id, ErrNotFound) }
    r.items = append(r.items[:i], r.items[i+1:]...) // Day 3 的删除写法
    return nil
}

// 编译期契约检查：*MemRepo 必须满足 TodoRepository，漏方法就编译失败
var _ TodoRepository = (*MemRepo)(nil)

var _ TodoRepository = (*MemRepo)(nil) 是 Go 社区标准做法：用空标识符做编译期契约检查，既不创建实例，又能在改接口时立刻暴露"哪个实现漏了方法"。试着把接口里的 List() 改名成 All() 再 go run .，报错长这样：repo.go:52:18: cannot use (*MemRepo)(nil) (value of type *MemRepo) as TodoRepository value in variable declaration: *MemRepo does not implement TodoRepository (missing method All)。

为什么连 List 都要带 error？内存版永远不会失败，返回 nil 就行；但文件版要读磁盘、第 3 周的 MySQL 版要连库，一定会失败。接口把"可能失败"提前写进签名，明天的 handler 就能一句 if err != nil { s.fail(w, err); return } 通吃所有实现——这就是"面向接口编程"落到手上的样子。

import (

type Todo struct {

type Logger interface { Log(msg string) }
type consoleLogger struct { prefix string }

func useLogger(l Logger) { l.Log("hello") } // 调用方不知道具体类型

// ③ 三层错误包装（讲义 4 的骨架，换成你自己的话术）
var ErrNotFound = errors.New("todo not found")

func readFromDisk(path string) ([]byte, error) {
    if path == "" { return nil, fmt.Errorf("open %q: %w", path, ErrNotFound) } // ← 把 %w 改成 %v 再跑
    return []byte("[]"), nil
}
func loadTodos(path string) (string, error) {
    data, err := readFromDisk(path)
    if err != nil { return "", fmt.Errorf("load todos: %w", err) }
    return string(data), nil
}

func main() {
    t := Todo{ID: 1, Title: "写 Stringer"}
    fmt.Println(t)   // 值：走你的 String()
    fmt.Println(&t)  // 指针：值方法自动进指针方法集，同样走 String()
    fmt.Println([]Todo{{ID: 2, Title: "slice 里也走"}, {ID: 1, Title: "写 Stringer", Done: true}})
    useLogger(consoleLogger{prefix: "[info]"})

    _, err := loadTodos("")

}

练习 3：仓储接口双实现 + 一套验收断言（45 分钟）

todo-api/filerepo.go（package main，不写 func main）package main

type FileRepo struct {
    inner *MemRepo // 组合：复用内存实现的逻辑，只做持久化增强
    path  string
}

func NewFileRepo(path string) (*FileRepo, error) {
    todos, err := LoadTodos(path) // store.go 里 Day 3 的那个函数，同包直接调
    if err != nil { return nil, err }
    return &FileRepo{inner: &MemRepo{items: todos, nextID: NextID(todos)}, path: path}, nil
}

// 只读方法直接转发；写方法先改内存、成功再整体落盘
func (f *FileRepo) List() ([]Todo, error)     { return f.inner.List() }
func (f *FileRepo) Get(id int) (Todo, error) { return f.inner.Get(id) }
func (f *FileRepo) Create(t string) (Todo, error)  { return f.write(f.inner.Create(t)) }
func (f *FileRepo) Complete(id int) (Todo, error) { return f.write(f.inner.Complete(id)) }
func (f *FileRepo) Delete(id int) error {
    if err := f.inner.Delete(id); err != nil { return err }
    items, err := f.inner.List()
    if err != nil { return err }
    return SaveTodos(f.path, items)
}
func (f *FileRepo) write(t Todo, err error) (Todo, error) {
    if err != nil { return Todo{}, err } // 内存那步就失败了，当然不落盘
    items, err := f.inner.List()
    if err != nil { return t, err }
    return t, SaveTodos(f.path, items) // 一次调用转手传下去，这是 Go 里很常见的"包装写回"
}

// 契约检查：FileRepo 也必须满足同一个接口
var _ TodoRepository = (*FileRepo)(nil)
todo-api/acceptance.go（package main，不写 func main）package main

import (
    "errors"
    "fmt"
)

func runAcceptance(repo TodoRepository) error {
    before, err := repo.List()
    if err != nil { return err }
    t, err := repo.Create("接口练习")
    if err != nil { return err }
    after, err := repo.List()
    if err != nil { return err }
    if len(after) != len(before)+1 { return errors.New("Create 后数量不对") }
    if got, err := repo.Complete(t.ID); err != nil || !got.Done { return errors.New("Complete 失败") }
    if err := repo.Delete(t.ID); err != nil { return err }
    if _, err := repo.Get(t.ID); !errors.Is(err, ErrNotFound) { return fmt.Errorf("应报 ErrNotFound，实际 %v", err) }
    if _, err := repo.Create("  "); !errors.Is(err, ErrEmptyTitle) { return errors.New("空标题应报错") }
    return nil
}
