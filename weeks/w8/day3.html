知识讲义

1. userInfo.role 进 store：扩展第 6 周的 useAuth

第 6 周的 useAuth 只存了 token。现在登录接口返回的 data.user 里已经带 role（D1 改了 JSON），把它一起收进来即可。role 数字与后端 model.Role 对齐：0=user，1=admin。

src/auth/useAuth.jsximport { createContext, useContext, useState, useCallback } from "react"
import api from "../api"

const AuthCtx = createContext(null)
export const ROLE_USER = 0
export const ROLE_ADMIN = 1

export function AuthProvider({ children }) {
  // 刷新页面后从 localStorage 恢复；user 里必须带 role
  const [user, setUser] = useState(() => {
    try { return JSON.parse(localStorage.getItem("user") || "null") }
    catch { return null }
  })

  const login = useCallback(async (username, password) => {
    const { data } = await api.post("/api/v1/auth/login", { username, password })
    localStorage.setItem("accessToken", data.data.accessToken)
    localStorage.setItem("refreshToken", data.data.refreshToken)
    localStorage.setItem("user", JSON.stringify(data.data.user)) // ← 含 role
    setUser(data.data.user)
  }, [])

  const logout = useCallback(() => {
    ["accessToken", "refreshToken", "user"].forEach(k => localStorage.removeItem(k))
    setUser(null)
  }, [])

  const value = { user, isAdmin: user?.role >= ROLE_ADMIN, login, logout }
  return <AuthCtx.Provider value={value}>{children}</AuthCtx.Provider>
}

export function useAuth() {
  const ctx = useContext(AuthCtx)
  if (!ctx) throw new Error("useAuth 必须在 AuthProvider 内使用")
  return ctx
}

localStorage 里的 role 是用户可随意篡改的快照（开 DevTools 把 0 改成 1 就能"自封管理员"）。它只决定「看不看得见入口」；管理员页发出的每个请求都会被后端 RequireRole 用真 token 里的 role 把关——这就是为什么 D1 坚持服务端也要判。

2. ProtectedRoute 加 role 参数

第 6 周的路由守卫只判「有没有登录」。升级后签名变成 <ProtectedRoute role={1}>：未登录 → 跳登录页（登录后送回原页）；登录了但角色不够 → 渲染 403 提示（不要跳回首页装没发生，明确报错对调试和用户都更诚实）。

src/auth/ProtectedRoute.jsximport { useLocation, Navigate, Link } from "react-router-dom"
import { useAuth } from "./useAuth"

export default function ProtectedRoute({ children, role }) {
  const { user } = useAuth()
  const location = useLocation()

  if (!user) {
    // 记录来路：登录成功后跳回来，第 6 周已做，保持不动
    return <Navigate to="/login" state={{ from: location }} replace />
  }
  if (role !== undefined && user.role < role) {
    return (
      <div className="denied">
        <h2>403 · 无权限</h2>
        <p>这个页面只对管理员开放。<Link to="/">回到首页</Link></p>
      </div>
    )
  }
  return children
}
src/App.jsx（路由表节选）<Route path="/editor/:id?" element={
  <ProtectedRoute><ArticleEditor /></ProtectedRoute>        // 只要登录} />
<Route path="/admin" element={
  <ProtectedRoute role={1}><AdminPanel /></ProtectedRoute> // 必须是管理员} />

对比 JS 前端老思路：以前「藏路由」就完事；现在你要能说出三层防御——菜单不显示（体验）→ 路由守卫 403（体验）→ 后端 RequireRole（安全）。前两层被绕过无所谓，第三层才是墙。

4. 管理员页面：用户表格 + 强制删文

完全复用第 6 周列表页的写法（useEffect 拉数据 + map 渲染 + 按钮调接口），只是接口换成 D1 的 /admin/*。被 403 时后端返回 code:1003，axios 拦截器已有统一错误提示，页面不用特判。

src/pages/AdminPanel.jsximport { useEffect, useState, useCallback } from "react"
import api from "../api"

export default function AdminPanel() {
  const [users, setUsers] = useState([])
  const [articles, setArticles] = useState([])
  const [stats, setStats] = useState(null)

  const load = useCallback(async () => {
    const [u, a, s] = await Promise.all([
      api.get("/api/v1/admin/users", { params: { page: 1, size: 20 } }),
      api.get("/api/v1/articles", { params: { page: 1, size: 20 } }),
      api.get("/api/v1/admin/stats"),
    ])
    setUsers(u.data.data.list)
    setArticles(a.data.data.list)
    setStats(s.data.data)
  }, [])

  useEffect(() => { load().catch(() => {}) }, [load])

  async function forceDelete(id) {
    if (!confirm("确定强制下架这篇文章？（软删除，可恢复）")) return
    await api.delete(`/api/v1/admin/articles/${id}`)
    load() // 删完重新拉，比本地 splice 更不易出 bug
  }

  return (
    <div className="admin">
      <h1>管理后台</h1>
      {stats && <p>用户 {stats.users} · 文章 {stats.articles} · 评论 {stats.comments}</p>}

      <h2>用户</h2>
      <table border="1" cellPadding="6">
        <thead><tr><th>ID</th><th>用户名</th><th>角色</th><th>注册时间</th></tr></thead>
        <tbody>
          {users.map(u => (
            <tr key={u.ID}>
              <td>{u.ID}</td><td>{u.username}</td>
              <td>{u.role === 1 ? "管理员" : "普通用户"}</td>
              <td>{new Date(u.CreatedAt).toLocaleDateString()}</td>
            </tr>
          ))}
        </tbody>
      </table>

      <h2>文章管理</h2>
      <ul>
        {articles.map(a => (
          <li key={a.ID}>
            <Link to={`/article/${a.ID}`}>{a.title}</Link>（{a.author?.username}）
            <button className="danger" onClick={() => forceDelete(a.ID)}>强制下架</button>
          </li>
        ))}
      </ul>
    </div>
  )
}

5. 评论查漏（一）：后端 Comment 模型与接口

规划第 8 周「保留功能」里写了评论——如果第 3~4 周你没做，现在补，一次到位。一对多复用上第 3 周：Article → []Comment、Comment belongsTo Author。注意 JSON 里带出 author，前端才知道「谁写的、我能不能删」。

internal/model/comment.gotype Comment struct {
    gorm.Model
    Content   string `gorm:"type:text;notNull" json:"content"`
    ArticleID uint   `gorm:"index;notNull" json:"article_id"`
    AuthorID  uint   `gorm:"index;notNull" json:"author_id"`
    Author    User   `gorm:"foreignKey:AuthorID" json:"author"`
}
internal/handler/comment.go// ListComments GET /api/v1/articles/:id/comments —— 游客可读
func (h *CommentHandler) ListComments(c *gin.Context) {
    var list []model.Comment
    h.db.Where("article_id = ?", c.Param("id")).
        Preload("Author").Order("id DESC").Find(&list)
    response.OK(c, list)
}

// CreateComment POST /api/v1/articles/:id/comments —— 需登录
func (h *CommentHandler) CreateComment(c *gin.Context) {
    var req struct {
        Content string `json:"content" binding:"required,max=500"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, 400, 1004, "评论内容 1–500 字")
        return
    }
    cm := model.Comment{
        Content: req.Content,
        ArticleID: uint(h.mustUint(c.Param("id"))),
        AuthorID:  c.GetUint("uid"),
    }
    if err := h.db.Create(&cm).Error; err != nil {
        response.Fail(c, 400, 1004, "文章不存在或提交失败")
        return
    }
    h.db.Preload("Author").First(&cm, cm.ID)
    response.Created(c, cm)
}

// DeleteComment DELETE /api/v1/comments/:id —— 作者本人或管理员，二选一
func (h *CommentHandler) DeleteComment(c *gin.Context) {
    var cm model.Comment
    if err := h.db.First(&cm, c.Param("id")).Error; err != nil {
        response.Fail(c, 404, 1001, "评论不存在")
        return
    }
    isOwner := cm.AuthorID == c.GetUint("uid")
    isAdmin := c.GetInt8("role") == 1 // RBAC 与所有权在同一接口合体
    if !isOwner && !isAdmin {
        response.Fail(c, 403, 1003, "只能删除自己的评论")
        return
    }
    h.db.Delete(&cm)
    response.OK(c, gin.H{"deleted": cm.ID})
}
迁移 SQL（与 D1 同套路）CREATE TABLE IF NOT EXISTS comments (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME(3) NULL,
  updated_at DATETIME(3) NULL,
  deleted_at DATETIME(3) NULL,
  content TEXT NOT NULL,
  article_id BIGINT UNSIGNED NOT NULL,
  author_id BIGINT UNSIGNED NOT NULL,
  INDEX idx_comments_article (article_id),
  INDEX idx_comments_deleted (deleted_at)
);

6. 评论查漏（二）：前端评论区组件

src/components/Comments.jsximport { useEffect, useState } from "react"
import { Link } from "react-router-dom"
import api from "../api"
import { useAuth } from "../auth/useAuth"

export default function Comments({ articleId }) {
  const { user, isAdmin } = useAuth()
  const [list, setList] = useState([])
  const [text, setText] = useState("")

  const load = () =>
    api.get(`/api/v1/articles/${articleId}/comments`).then(r => setList(r.data.data))
  useEffect(() => { load() }, [articleId])

  async function submit(e) {
    e.preventDefault()
    if (!text.trim()) return
    await api.post(`/api/v1/articles/${articleId}/comments`, { content: text.trim() })
    setText(""); load()
  }

  async function remove(id) {
    if (!confirm("删除这条评论？")) return
    await api.delete(`/api/v1/comments/${id}`)   // 后端判断 owner 或 admin
    load()
  }

  return (
    <section className="comments">
      <h3>评论（{list.length}）</h3>
      {user ? (
        <form onSubmit={submit}>
          <textarea value={text} rows={3} maxLength={500}
            onChange={e => setText(e.target.value)} placeholder="友善发言…" />
          <button disabled={!text.trim()}>发布</button>
        </form>
      ) : <p><Link to="/login">登录</Link> 后参与讨论</p>}

      <ul>
        {list.map(cm => (
          <li key={cm.ID}>
            <b>{cm.author?.username}</b>：{cm.content}
            {(user?.ID === cm.author_id || isAdmin) && (
              <button className="link" onClick={() => remove(cm.ID)}>删除</button>
            )}
          </li>
        ))}
      </ul>
    </section>
  )
}

「作者能删自己的、管理员能删所有」这条规则后端已经实现了，前端的显示条件只是把不该出现的按钮藏起来。哪怕用户改前端强显删除按钮去调接口，不是他的评论照样吃 403。

今日练习

练习 1：核心验收——非管理员手敲 /admin URL（30 分钟）

用普通账号登录，地址栏直接输入 http://localhost:5173/admin。预期：不跳登录页（人已登录）、不渲染后台，而是出现你的 403 提示。再用管理员登录：正常进后台；退出后访问：被重定向到 /login。

参考答案：三种状态分别对应 ProtectedRoute 的三个分支。若普通用户能进 /admin，检查两点：路由表是否传了 role={1}；登录后 localStorage 的 user 是否真的带 role（旧登录态是升级前存的，退出重登一次再看，这个坑今天必踩一次）。

练习 2：绕过前端直捣接口（20 分钟）

普通用户登录状态下，DevTools 把 localStorage 的 user.role 改成 1 —— 后台入口出现了！点进去、强制下架一篇文章。预期：接口返回 403 / code 1003，删除失败。

参考答案：这就是「前端权限是纸糊的」实锤演示；token 里的 role 才是后端看到的身份。把这次体验写进 README 的权限设计一节，面试可直接讲。

练习 3：评论闭环（60 分钟）

跑迁移建 comments 表 → 挂三个评论路由 → 详情页嵌 <Comments articleId={id}/>。验收：游客可看；A 发评论；B 看 A 的评论旁无删除按钮；A 自己和管理员有。管理员删任何评论均 200。

练习 4（选做）：把 D2 头像接进 Navbar（30 分钟）

个人信息页放 <ImageUpload kind="avatar" onDone={url => { localStorage 更新 user.avatar; 触发重渲染 }}/>，Navbar 右上角用 <img> 显示头像。注意 onDone 后要同步 localStorage 里的 user 快照。

常见坑

坑现象正确姿势
升级前登录的旧 user 快照没有 role管理员也不见后台入口，守卫判定 undefined < 1 为 false 还处处诡异改字段后强制重新登录；读快照时兜底 user.role ?? 0
ProtectedRoute 把权限不足重定向回首页用户以为页面坏了，问题无法暴露渲染明确的 403 文案，别静默跳转
只在菜单里藏入口，不加路由守卫手敲 /admin 直接进，接口 403 刷满页面守卫 + 后端双保险，缺一层都不算完成
Comment 忘 Preload("Author")前端 cm.author.username 报错白屏接口返回带作者；前端渲染再写 cm.author?.username
软删除评论后列表计数不同步stats 里 comments 数与列表对不上GORM Count 自动排除软删行；手写 SQL 才要加 deleted_at IS NULL
