3. 回跳方案全站只有一条。第 6 周定的是 /login?redirect=编码后的 pathname+search，登录后 decodeURIComponent 再 navigate。今天新增的 403 分支不搞 state.from，未登录分支保持不动。

知识讲义

1. role 进 store：useAuth.js 的三处增量

第 6 周的 useAuth() 从 localStorage 读 user。D1 之后登录接口的 data.user 里已经带 role（0=user 1=admin，和后端 model.Role 逐字对齐），所以写入侧一行都不用改——saveSession() 是整个 user 对象塞进去的，role 顺路就来了。要动的只有读取侧：把 role 提成显眼的能力，外加兑现 D2 欠的那个 refreshUser()。

user?.role ?? ROLE_USER 这行不是洁癖：D1 之前登录的用户，localStorage 里的 user 根本没有 role 键，undefined < 1 结果是 false——守卫会放行，管理员页渲染出来再被接口 403 打个满脸。兜底成 0 之后行为才对：没角色 = 普通用户。今天必踩的第二个坑是「升级前登录的旧会话」，解法只有退出重登一次（练习 1 会让你两种都撞一遍）。

localStorage 里的 role 是用户可随意篡改的快照（DevTools 把 0 改成 1 就能「自封管理员」）。它只决定「看不看得见入口」；页面发出的每个请求都会被后端 RequireRole 用真 token 里的 role 把关——这正是 D1 坚持服务端也要判的原因。D2 那句话在这里同样成立：本地快照 vs 服务端真值的漂移，和第 1 天 JWT 里的 role 是同一类问题。

2. ProtectedRoute 加 role 参数（diff，不是重写）

第 6 周的守卫只判「有没有 token」。今天加第二个可选参数 role：传了就要比等级。两条分支的行为刻意不同——未登录 → 跳登录页（带 redirect，登录完送回原页）；已登录但角色不够 → 就地渲染 403。别把权限不足也重定向回首页：那是把 bug 藏起来，用户以为页面坏了，你查半小时。

src/pages/AdminPanel.jsx（新文件：整份复制即可跑）import { useEffect, useState, useCallback } from 'react'
import { Link } from 'react-router-dom'
import { listUsers, forceDeleteArticle, adminStats } from '../api/admin'
import { listArticles } from '../api/article'   // 第 6 周就有，直接复用

export default function AdminPanel() {
  const [users, setUsers] = useState([])
  const [articles, setArticles] = useState([])
  const [stats, setStats] = useState(null)
  const [busy, setBusy] = useState(false)
  const [err, setErr] = useState(null)

  const load = useCallback(async () => {
    try {
      // 四个请求并发；后端 PageResult 已经把 list 包好，所以取的是 res.list（没有第二层 .data）
      const [u, a, s] = await Promise.all([

        listArticles({ page: 1, size: 20 }),

      setErr(e.message)          // code≠0 时拦截器 reject，带的就是后端 message；403 会走到这里

  useEffect(() => { load() }, [load])

    setBusy(true)
    try {
      await forceDeleteArticle(id)
      load()                     // 删完重新拉，比本地 splice 更不易出 bug
    } catch (e) { setErr(e.message) } finally { setBusy(false) }
  }

  return (
    <div className="admin" style={{ padding: '20px' }}>
      <h1>管理后台</h1>
      {err && <p className="err">{err}</p>}
      {stats && <p>用户 {stats.users} · 文章 {stats.articles} · 评论 {stats.comments}</p>}

      <h2>用户</h2>

        <tbody>
          {users.map((u) => (
            <tr key={u.id}>                          // ← 第 5 节 DTO 之后才是小写 id
              <td>{u.id}</td>
              <td>{u.username}</td>
              <td>{u.nickname || '—'}</td>
              <td>{u.role === 1 ? '管理员' : '普通用户'}</td>
              <td>{new Date(u.created_at).toLocaleDateString()}</td>
            </tr>
          ))}
        </tbody>
      </table>

      <h2>文章管理</h2>
      <ul>
        {articles.map((a) => (
          <li key={a.id}>
            <Link to={`/post/${a.id}`}>{a.title}</Link>（{a.user?.nickname}）
            <button disabled={busy} onClick={() => forceDelete(a.id)}>强制下架</button>
          </li>
        ))}
      </ul>
    </div>
  )
}

三处「今天最容易照抄错的」：① /post/${a.id} ——详情页路由是 /post/:id，不是 /article/:id；② 文章对象里作者键是 user（第 3 周起的 json tag 就是 json:"user,omitempty"），不是 author，只有评论里的作者才叫 author（下一节自己定义的）；③ 403 时后端返回 code:1003，拦截器直接 reject，所以页面只要 setErr，不用自己判状态码。

这条坑值得单独记一句：迁移编号是「计划号」，不是「先来后到的号」。本周两对迁移（000005 评论、000006 头像）在真实团队里会写在同一个 PR 里一次合入，migrate 自然按 5→6 顺序跑；我们是把它拆成两天教，才出现「6 已跑、5 后补」这种顺序倒置。上面那三行 down 1 / force 4 / up 是解法，但只在本地或测试库做（force 是改数字不改结构，生产上乱 force 会让记录和真实结构脱钩，那就是第 7 周 D4 讲的 dirty 事故现场）。生产要加评论表？新开一对 000007_add_comments，让编号继续单调递增。

不想动版本记录也有第二条路：把上面这段 SQL 直接贴进 docker compose exec mysql 执行，再 migrate force 6？不要——手敲 SQL 建完表又去改版本号，等于让 schema_migrations 从此说谎。照抄上面三步，或者干脆重开一个干净库（docker compose down -v 会连 mysql-data 一起删，练手库才许这么干）按 D1→D2→D3 的顺序把两对迁移一次跑齐。

// 形状照第 3 周的 ArticleRepository（List/Get/Create/Delete + 返回 total）。
type CommentRepository interface {
    ListByArticle(articleID uint, limit, offset int) ([]model.Comment, int64, error)
    Get(id uint) (model.Comment, error)
    Create(cm model.Comment) (model.Comment, error)
    Delete(id uint) error
}

// ErrCommentNotFound 哨兵：handler 用 errors.Is 认它，翻译成 404 + 1001（通用「资源不存在」）
var ErrCommentNotFound = errors.New("评论不存在")

type gormCommentRepo struct { db *gorm.DB }

func NewCommentRepo(db *gorm.DB) CommentRepository { return &gormCommentRepo{db: db} }

var _ CommentRepository = (*gormCommentRepo)(nil)   // 编译期接口断言，第 3 周的规矩

// ListByArticle 两条 SELECT：Count 一次 + 主表一次（Preload Author 再一条 IN，共 3 条）。
// 不 Preload 就成了 for 循环里逐条查作者的 N+1，第 3 周那笔账在这儿重算一遍没有意义。
func (r *gormCommentRepo) ListByArticle(articleID uint, limit, offset int) ([]model.Comment, int64, error) {
    var total int64
    if err := r.db.Model(&model.Comment{}).Where("article_id = ?", articleID).Count(&total).Error; err != nil {
        return nil, 0, err
    }
    list := []model.Comment{}
    err := r.db.Where("article_id = ?", articleID).
        Preload("Author").
        Order("id DESC").Limit(limit).Offset(offset).
        Find(&list).Error
    return list, total, err
}

func (r *gormCommentRepo) Get(id uint) (model.Comment, error) {
    var cm model.Comment
    err := r.db.Preload("Author").First(&cm, id).Error
    if errors.Is(err, gorm.ErrRecordNotFound) { return model.Comment{}, ErrCommentNotFound }
    return cm, err
}

func (r *gormCommentRepo) Create(cm model.Comment) (model.Comment, error) {
    if err := r.db.Create(&cm).Error; err != nil { return model.Comment{}, err }
    return cm, nil
}

func (r *gormCommentRepo) Delete(id uint) error {
    res := r.db.Delete(&model.Comment{}, id)   // 软删除：只打 deleted_at
    if res.Error != nil { return res.Error }
    if res.RowsAffected == 0 { return ErrCommentNotFound }
    return nil
}

// List 直接吃归一化好的 PageQuery（第 3 周 dto.PageQuery：Normalize + Offset 都是现成的）
func (s *CommentService) List(articleID uint, q dto.PageQuery) (dto.PageResult, error) {
    q.Normalize()
    list, total, err := s.repo.ListByArticle(articleID, q.Size, q.Offset())
    if err != nil {

    }
    return dto.NewPageResult(list, total, q), nil

// Create 顺序很讲究：先查文章（贵但能挡脏写），再清洗内容（免费的字符串活）。
// 文章不存在时把 repository 的 ErrArticleNotFound 原样抛出去，handler 翻成 404 + 2001。
func (s *CommentService) Create(articleID, uid uint, content string) (model.Comment, error) {
    if _, err := s.arts.Get(articleID); err != nil {
        return model.Comment{}, err
    }
    content = strings.TrimSpace(content)
    if content == "" {
        return model.Comment{}, ErrCommentEmpty
    }
    if utf8.RuneCountInString(content) > MaxCommentRunes {
        return model.Comment{}, ErrCommentLong
    }
    return s.repo.Create(model.Comment{Content: content, ArticleID: articleID, AuthorID: uid})
}

// Delete 「作者本人 或 管理员」：RBAC 与资源所有权第一次在同一个接口里合体。
// 这两行判定在 D4 会被抽成 internal/service/rbac.go 的纯函数 CanModerate 并补表驱动单测——
// 今天先照抄，D4 回来替换，别自己发明第三种写法。
func (s *CommentService) Delete(id, uid uint, role model.Role) error {
    cm, err := s.repo.Get(id)
    if err != nil {
        return err                       // ErrCommentNotFound 原样上抛
    }
    isOwner := cm.AuthorID == uid
    isAdmin := role == model.RoleAdmin
    if !isOwner && !isAdmin {
        return ErrNotAuthor
    }
    return s.repo.Delete(id)
}

utf8.RuneCountInString 而不是 len(content)：一个汉字 3 字节，按字节数限长的话中文用户 167 字就被拒。前端 maxLength={500} 数的是字符，两边都按字符算才对得上——这条属于「中日韩产品必踩」的通用坑，值得进你的 BUGLOG 模板。

func idParam(c *gin.Context, name string) (uint, bool) {
    id, err := strconv.ParseUint(c.Param(name), 10, 32)
    if err != nil || id == 0 {
        return 0, false
    }
    return uint(id), true

// currentRole 读 D1 的 Auth 中间件塞进 context 的角色。
// 全站基准只有这一处写法：键名 "role"，类型 model.Role；
// 拿不到（忘挂 Auth / 类型不匹配）就退回最小权限 RoleUser，绝不退回零值再碰运气。
func currentRole(c *gin.Context) model.Role {
    raw, exists := c.Get("role")
    if !exists {
        return model.RoleUser
    }
    role, ok := raw.(model.Role)
    if !ok {
        return model.RoleUser
    }
    return role
}

// Create POST /api/v1/articles/:id/comments —— 需登录（组上挂了 Auth）
func (h *CommentHandler) Create(c *gin.Context) {
    articleID, ok := idParam(c, "id")
    if !ok {
        response.Fail(c, http.StatusBadRequest, 1004, "id 必须是正整数")
        return
    }
    var req struct {
        Content string `json:"content" binding:"required,max=500"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, http.StatusBadRequest, 1004, "评论内容 1–500 字")
        return
    }
    cm, err := h.svc.Create(articleID, c.GetUint("userID"), req.Content)  // ← 键是 userID，不是裸 "uid"
    if err != nil {
        switch {
        case errors.Is(err, repository.ErrArticleNotFound):
            response.Fail(c, http.StatusNotFound, 2001, "文章不存在")
        case errors.Is(err, service.ErrCommentEmpty), errors.Is(err, service.ErrCommentLong):
            response.Fail(c, http.StatusBadRequest, 1004, err.Error())   // 文案直接透给前端
        default:
            response.Fail(c, http.StatusInternalServerError, 5000, "提交失败")
        }
        return
    }
    // 回读一次把 Author 带出来：刚 Create 的对象里 Author 是零值，前端拿不到「谁写的」
    full, err := h.svc.Get(cm.ID)
    if err != nil {
        response.Created(c, cm)   // 回读失败不影响「已经写进去了」这个事实，退而求其次
        return
    }
    response.Created(c, full)
}

}

练习 1：核心验收——非管理员手敲 /admin URL（30 分钟）

用普通账号登录，地址栏直接输入 http://localhost:5173/admin。预期：不跳登录页（人已登录）、不渲染后台，而是出现你的 403 提示。再用管理员登录：正常进后台；退出后再访问：被弹去 /login?redirect=%2Fadmin，登录后回到 /admin 而不是首页。

参考答案：三种状态分别对应 ProtectedRoute 的三个分支（无 token / 角色不够 / 通过）。若普通用户能进 /admin，检查两点：路由表是否传了 role={ROLE_ADMIN}；(user?.role ?? 0) < role 那行有没有被写成 user.role < role（老快照下 undefined 比较为 false，等于放行）。若管理员也被挡在 403 外，八成是登录态是升级前存的，退出重登一次再看——这个坑今天必踩一次。

练习 2：绕过前端直捣接口（20 分钟）

普通用户登录状态下，DevTools 把 localStorage 里 user 的 role 改成 1 —— 后台入口出现了！点进去、强制下架一篇文章。预期：接口返回 403 / code 1003，删除失败，页面顶部显示「权限不足」。

练习 3：评论闭环（60 分钟）

按第 7~10 节把评论三层抄齐并跑迁移 → 挂三条评论路由 → 详情页嵌 <Comments articleId={Number(id)} />。验收清单：游客能看到评论列表；A 发评论成功（201 且带回 author.nickname）；B 看 A 的评论旁没有删除按钮；A 自己删成功、管理员删任何一条都成功；docker compose --profile tools run --rm migrate version 是 6。
