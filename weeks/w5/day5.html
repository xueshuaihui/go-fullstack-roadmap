并发模式与本周项目：并发 URL 聚合 API

四天零件都齐了：goroutine 启动任务（D1）、channel 传结果（D2）、Mutex 保共享状态（D3）、context 管超时取消（D4）。今天把它们装成两个工业级模式——worker pool 与 errgroup——然后完成本周项目：一个并发 URL 聚合 API。验收标准已锁定为固定模拟延迟：5 个各 sleep 1 秒的端点，串行 ≈ 5s、并发 ≈ 1s，含 8 秒慢端点时 5 秒截停返回部分结果——不受网络波动影响，数字必须漂亮地稳定。

知识讲义

1. worker pool：任务分发流水线

问题：10000 个任务，无脑一人一个 goroutine 会把下游打挂。worker pool：固定开 N 个工人，任务从一条 channel 排队发出去，谁空闲谁领活。并发度被你钉死在 N，channel 满了生产者自动被背压减速（day2 学的）：

main.gopackage main

import (
    fmt
    sync
    time
)

func main() {
    jobs := make(chan int) // 任务队列（无缓冲：领一个发一个）
    var wg sync.WaitGroup

    for w := 1; w <= 3; w++ { // 固定 3 个工人
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            for j := range jobs { // jobs 被 close 且取空前一直领活
                fmt.Println("工人", id, "处理任务", j)
                time.Sleep(200 * time.Millisecond)
            }
        }(w)
    }

    go func() { // 生产者：投 10 个任务后关闸
        for i := 1; i <= 10; i++ { jobs <- i }
        close(jobs) // 忘 close：worker 的 for range 永不退出 → 死锁
    }()

    wg.Wait()
    fmt.Println("10 个任务全部消化")
}

JS 类比：你以前用 p-limit 包一层 await limit.add(() => fetch(...)) 控制并发数——worker pool 就是它的语言原生版，且工人是真并行的，CPU 密集任务也能吃满多核。

2. errgroup：带超时、带取消、带错误聚合的 WaitGroup

golang.org/x/sync/errgroup 是官方扩展库，把本周前四天揉成一个小结构体：

  • ≈ WaitGroup：g.Go(func() error) 起任务，g.Wait() 等全部收工；
  • 错误聚合：Wait() 返回第一个非 nil 错误，不用自己攒 err；
  • WithContext：返回的 ctx 在任一任务首次出错时自动取消——失败快跑，兄弟任务立刻停工，不烧无用资源；
  • SetLimit(n)：最多同时跑 n 个，天然的 worker pool 限流。
main.gopackage main

import (
    context
    errors
    fmt
    time

    "golang.org/x/sync/errgroup"
)

func main() {
    // WithContext：派生可自动取消的 ctx（再叠 5 秒总超时就是完整保险）
    parent, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    g, ctx := errgroup.WithContext(parent)
    g.SetLimit(2) // 同时最多 2 个任务在跑

    tasks := []struct{
        name string
        cost time.Duration
        fail bool
    }{
        {"查用户", 300 * time.Millisecond, false},
        {"查订单", 500 * time.Millisecond, true},  // 故意失败
        {"查推荐", 900 * time.Millisecond, false},
    }
    for _, t := range tasks {
        t := t // Go 1.22+ 可省略，老代码常见防共享写法
        g.Go(func() error {
            select {
            case <-time.After(t.cost):
                if t.fail {
                    return errors.New(t.name + " 炸了")
                }
                fmt.Println(t.name, "完成")
                return nil
            case <-ctx.Done(): // 别的任务出错 → g 的 ctx 被取消 → 我提前撤退
                fmt.Println(t.name, "被取消:", ctx.Err())
                return ctx.Err()
            }
        })
    }

    if err := g.Wait(); err != nil {
        fmt.Println("聚合结果: 任务组失败 -", err) // 返回首个错误
    } else {
        fmt.Println("全部成功")
    }
}

关键取舍：聚合场景别滥用 WithContext。今天的聚合 API 要的是"部分成功也算成功"，所以每个任务把错误写进自己的结果里、永远 return nil，errgroup 只当"并发调度+等待"用；如果换成"任何一页抓取失败整个报表作废"的场景，才让任务 return error 触发全员取消。两种语义都要会，选型看业务。

3. 本周项目：并发 URL 聚合 API —— 完整实现

新建目录 url-aggregator，go mod init url-aggregator 后执行 go get golang.org/x/sync/errgroup。全部代码就一个 main.go，直接复制可运行：

main.gopackage main

import (
    context
    encoding/json
    fmt
    log
    net/http
    time

    "golang.org/x/sync/errgroup"
)

// ---------- 数据结构 ----------
type reqBody struct {
    URLs []string json:"urls"
}

type urlResult struct {
    URL     string json:"url"
    Status  int    json:"status"
    Elapsed string json:"elapsed"
    Error   string json:"error,omitempty"
}

type aggResponse struct {
    Mode    string      json:"mode"
    TotalMS int64       json:"total_ms"
    Partial bool        json:"partial"
    Results []urlResult json:"results"
}

// ---------- 模拟慢端点：固定 sleep，排除网络波动 ----------
func mockSlow(w http.ResponseWriter, r *http.Request) {
    d, err := time.ParseDuration(r.URL.Query().Get("d"))
    if err != nil {
        d = 1 * time.Second
    }
    select {
    case <-time.After(d):
        fmt.Fprintf(w, "pong after %v", d)
    case <-r.Context().Done(): // 聚合器 5s 截停后，这里立刻感知
        log.Println("mock-slow 被取消:", d)
    }
}

// ---------- 抓取单个 URL：错误写进结果，绝不 panic ----------
func fetchOne(ctx context.Context, url string) urlResult {
    start := time.Now()
    res := urlResult{URL: url}

    req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
    if err != nil {
        res.Error, res.Elapsed = err.Error(), time.Since(start).String()
        return res
    }
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        res.Error, res.Elapsed = err.Error(), time.Since(start).String()
        return res
    }
    defer resp.Body.Close()
    res.Status = resp.StatusCode
    res.Elapsed = time.Since(start).Round(10 * time.Millisecond).String()
    return res
}

// ---------- POST /aggregate?mode=serial|concurrent ----------
func aggregate(w http.ResponseWriter, r *http.Request) {
    var body reqBody
    if err := json.NewDecoder(r.Body).Decode(&body); err != nil || len(body.URLs) == 0 {
        http.Error(w, `{"error":"需要 JSON: {\"urls\": [...]}"}`, http.StatusBadRequest)
        return
    }

    ctx, cancel := context.WithTimeout(r.Context(), 5*time.Second) // 本周验收：5 秒总预算
    defer cancel()
    start := time.Now()
    results := make([]urlResult, len(body.URLs)) // 预分配按下标写：不重叠，免锁（D2 练习 2 结论）

    mode := r.URL.Query().Get("mode")
    if mode == "serial" {
        for i, u := range body.URLs {
            results[i] = fetchOne(ctx, u) // 串行：一个跑完才轮到下一个
        }
    } else {
        g, _ := errgroup.WithContext(ctx)
        g.SetLimit(5) // 并发上限 5，防止 URL 列表过长打挂下游
        for i, u := range body.URLs {
            i, u := i, u
            g.Go(func() error {
                results[i] = fetchOne(ctx, u)
                return nil // 聚合语义：错误进 results，不触发全员取消
            })
        }
        _ = g.Wait()
    }

    partial := ctx.Err() != nil
    for _, res := range results {
        if res.Error != "" {
            partial = true
        }
    }
    if partial { // 超时提示：让调用方知道结果不完整
        for i := range results {
            if results[i].Error == "" && results[i].Status == 0 {
                results[i].Error = "context deadline exceeded（5s 内未完成）"
            }
        }
    }

    w.Header().Set("Content-Type", "application/json; charset=utf-8")
    json.NewEncoder(w).Encode(aggResponse{
        Mode:    map[bool]string{true: "serial", false: "concurrent"}[mode == "serial"],
        TotalMS: time.Since(start).Milliseconds(),
        Partial: partial,
        Results: results,
    })
}

func main() {
    mux := http.NewServeMux()
    mux.HandleFunc("/mock-slow", mockSlow)
    mux.HandleFunc("/aggregate", aggregate)
    log.Println("listening :8080")
    log.Fatal(http.ListenAndServe(":8080", mux))
}

读代码时对照本周日历：errgroup=今天的 D5；预分配按下标写=D2 练习 2；fetchOne 检查 ctx=D4；mockSlow 的 r.Context().Done()=D4 练习 2。这就是"周中零件周末装配"的本手册节奏。

今日练习

练习 1：手搓 worker pool（必做，25 分钟）

照讲义第 1 节默写 worker pool：10 个任务、3 个工人。完成后改造：① 把 close(jobs) 删掉，观察死锁报错原文；② 工人数改成 10，对比 200ms/任务的总耗时变化（3 人≈0.8s 级 vs 10 人≈0.2s 级），体会儿"并发度是旋钮不是越多越好"。

参考答案

讲义第 1 节代码即完整答案。删 close 后约 1 分钟得到 fatal error: all goroutines are asleep - deadlock!（3 个工人在等 jobs，生产者在等被取走，main 在等 wg.Wait）。改 10 工人后 10 个任务同时开跑，总耗时从 ≈800ms 降到 ≈200ms——但注意这是 sleep 模拟 IO，换成 CPU 计算则工人超过核数后不再线性加速。

练习 2：errgroup 三种语义对比（25 分钟）

基于讲义第 2 节做三组实验：① 原样跑，记录"查推荐被取消"日志与 Wait() 返回的首错；② 把 fail 全改 false，确认"全部成功"；③ 把 g.Go 里改成"错误写进结果、永远 return nil"，确认 WithContext 不再取消任何人——亲手验证"何时 return error、何时吞 error"的选型。

参考答案要点

①：500ms"查订单"炸，900ms 的"查推荐"在 500ms 时刻被取消（提前 400ms 收工），g.Wait() 返回 查订单 炸了。③：三个任务各自完成，Wait() 返回 nil，错误信息都在你自己收集的结构里。今天项目用的是 ③ 的语义（聚合要部分结果），批处理结算类场景才用 ①（一错全撤）。

练习 3：装配本周项目并记录三组数字（必做，60 分钟）

照讲义第 3 节建好 url-aggregator 并跑通三组验收，把数字写进笔记：① 5 个 d=1s 并发（期望 total_ms 约 1000–1200）；② 同请求 ?mode=serial（期望约 5000）；③ 把其中一个换成 d=8s（期望约 5000 截停、partial:true、8s 项带 error，且终端 1 出现"mock-slow 被取消"日志）。全程用 go run -race . 启动，确认零 WARNING。

排错提示
  • 并发模式耗时接近 5s → 多半把 g.Go(func() error {...}) 误写成末尾带调用的 g.Go(func() error {...}())：后者会先同步跑完再把返回值交给 g.Go，等于串行。
  • 超时场景 8s 项 error 为空 → 确认 ctx 传进了 http.NewRequestWithContext（D4 头号坑）。
  • -race 报 results 切片竞态 → 检查是否有人绕过下标去 append。
练习 4：给聚合器加一个"健康检查+计数"（选做，20 分钟）

加全局 atomic.Int64 记录 aggregate 被调次数，新增 GET /stats 返回 {"calls": N}；再想想：calls 若改用普通 int++，-race 会说什么？（D3 练习 1 的镜像题）

参考答案

var calls atomic.Int64（Go 1.19+ 方法式 API），aggregate 开头 calls.Add(1)，/stats 里 fmt.Fprintf(w, "{"calls": %d}", calls.Load())。改成裸 int++ 后并发请求 aggregate 时 -race 立刻报 WARNING: DATA RACE，两次 Write 指向同一 calls 地址——D3 学的报告逐行对应上了。

常见坑（新手 100% 会踩）

坑现象正确姿势
ctx 没传到底5 秒超时形同虚设：handler 到点了，http.Get 裸调用还在傻等 8 秒，接口 8 秒后才返回抓取必须走 http.NewRequestWithContext(ctx, ...)；GORM 用 db.WithContext(ctx)。超时不生效先全仓搜裸 http.Get
聚合场景误用 errgroup 首错取消一个 URL 失败，其余结果全变 context canceled，partial 响应里一半是取消噪音聚合语义任务 return nil、错误进结果字段；只有"一错全撤"的业务才让 error 上抛
忘 close 任务 channelworker 的 for range 永不结束，wg.Wait 死等 → 全局死锁 fatal error生产者投完必关闸；多生产者用 WaitGroup+看守 goroutine 统一关（D2 练习 1）
并发结果用 append 不加锁结果偶发缺条、错乱，-race 高亮 append 行预分配 make([]T, n) 按下标写（互不重叠免锁），或老老实实 Mutex
worker 数拍脑袋开 5000 并发打下游，对面连接池爆掉、超时雪崩，比串行还慢SetLimit/工人数按下游承受力设（常见 4–32），并配重试与熔断意识
