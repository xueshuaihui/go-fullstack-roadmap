第 3 周总结：把 API 接到真实的数据库上

这一周你完成了从「内存玩具」到「有持久化的后端」的跨越：Docker Compose 起 MySQL、GORM 建模与迁移、关联查询、分页搜索、事务与软删除。本周验收的核心不是功能多，而是「重启不丢」和「关联/分页正确」。下面的清单请逐条用 curl 跑一遍，全部通过再进第 4 周。

模型定义全文（含关联）

internal/model/user.gopackage model

import gorm.io/gorm

// gorm.Model = ID + CreatedAt + UpdatedAt + DeletedAt（自带软删除）
type User struct {
    gorm.Model

    Name     string `gorm:"column:name;type:varchar(64);size:64;not null" json:"name"`
    Email    string `gorm:"type:varchar(128);size:128;not null;uniqueIndex" json:"email"`
    Password string `gorm:"type:varchar(255);not null" json:"-"`

    Articles []Article `gorm:"foreignKey:UserID" json:"articles,omitempty"`
}
internal/model/article.gopackage model

import gorm.io/gorm

type Article struct {
    gorm.Model

    Title   string `gorm:"type:varchar(128);size:128;not null;index" json:"title"`
    Content string `gorm:"type:text" json:"content"`
    Views   int    `gorm:"not null;default:0" json:"views"`

    // 一对多：外键 articles.user_id 指向 users.id
    UserID uint `gorm:"type:bigint unsigned;not null;index" json:"user_id"`
    User   User `gorm:"foreignKey:UserID;references:ID" json:"user,omitempty"`

    // 多对多：中间表 article_tags(article_id, tag_id)
    Tags []Tag `gorm:"many2many:article_tags;" json:"tags,omitempty"`
}
internal/model/tag.gopackage model

import (
    time
)

type Tag struct {
    ID        uint      `gorm:"primaryKey" json:"id"`
    Name      string    `gorm:"type:varchar(32);size:32;not null;uniqueIndex" json:"name"`
    CreatedAt time.Time `json:"created_at"`
}

func (Tag) TableName() string { return "tags" }
cmd/server/main.go（装配与迁移）package main

import (
    log

    "github.com/gin-gonic/gin"

    "goblog/internal/config"
    "goblog/internal/handler"
    "goblog/internal/model"
    "goblog/internal/repository"
    "goblog/internal/service"
)

func main() {
    db, err := repository.New(config.DSN)      // gorm.Open + 连接池 + Ping
    if err != nil {
        log.Fatalf("init db: %v", err)
    }
    if err := db.AutoMigrate(&model.User{}, &model.Tag{}, &model.Article{}); err != nil {
        log.Fatalf("migrate: %v", err)
    }

    userRepo := repository.NewGormUserRepo(db)       // ← 上周这里是 NewMemUserRepo()
    artRepo := repository.NewGormArticleRepo(db)
    userSvc := service.NewUserService(userRepo, artRepo)
    artSvc := service.NewArticleService(artRepo, userRepo)

    r := gin.Default()
    r.GET("/health", func(c *gin.Context) { c.JSON(200, gin.H{"status": "ok"}) })

    h := handler.New(userSvc, artSvc)
    r.GET("/users", h.UserList)
    r.POST("/users", h.UserCreate)
    r.GET("/users/:id", h.UserDetail)
    r.PUT("/users/:id", h.UserUpdate)
    r.DELETE("/users/:id", h.UserDelete)
    r.GET("/users/:id/articles", h.ArticlesByUser)

    r.GET("/articles", h.ArticleList)          // page/size/keyword/sort/tag/user_id
    r.POST("/articles", h.ArticleCreate)        // 事务：建文章 + 挂标签
    r.GET("/articles/:id", h.ArticleDetail)      // Preload Tags + User
    r.PUT("/articles/:id", h.ArticleUpdate)
    r.DELETE("/articles/:id", h.ArticleDelete)   // 软删除

    log.Fatal(r.Run(":8080"))
}

从内存版改造成 MySQL 版的步骤（迁移清单）

#改造动作注意点
1模型从 struct + 手写 ID 自增 换成带 tag 的 GORM 模型（internal/model）给每个字段补 json tag，密码 json:"-"，否则响应结构一变前端就崩
2repository 接口保持不变，新增 GORM 实现（NewGormUserRepo(db) UserRepo）构造函数返回接口类型，编译器立刻能查出你漏实现了哪个方法
3把「内存里用 for 循环过滤」的查询换成 Where；「切片追加」换成 Createservice 层不要出现 *gorm.DB，否则接口又漏出实现了
4「找不到就返回 nil, nil」的语义改成 ErrXxxNotFound + errors.Is让 handler 能区分 404 与 500，这是本周错误口径的关键升级
5删掉 map + RWMutex 的并发保护代码内存版才需要自己加锁；并发正确性交给数据库与事务（D5）
6main.go 里换装配，保留旧实现在文件里注释一行以后写测试时可以用内存版或 sqlite 桩做「双实现对照」

做完这六步你会直观地看到：变的只有 repository 和 model，service/handler 几乎零改动。这就是第 2 周分层与第 1 周接口设计的回报——依赖倒置不是术语，是「换实现不重写」这件事本身。

卡点复盘（本周日必写，10 分钟）

高频卡点典型症状标准解法
容器起来了但 Go 连不上connect: connection refused / invalid DSN / Access denied for user依次查：docker compose ps 是否 healthy → DSN 端口是否与 ports 左侧一致 → 账号密码是否为 MYSQL_USER/MYSQL_PASSWORD（不是 root 的）→ 库名是否存在
时间差 8 小时created_at 比实际早 8 小时，排序/日期筛选诡异DSN 加 parseTime=True&loc=Local；或统一容器时区。全项目只选一种
数据莫名消失昨天插的数据今天查不到检查 volumes 是否被注释；是否执行过 down -v；是否连到了另一套 MySQL（Homebrew 那个）
列表接口 800ms日志几十条 SELECTN+1 → 改 Preload；排序列加索引；SELECT * 换成需要的列
AutoMigrate 不生效改了 tag 表结构没变它不删列不改类型；手工 ALTER TABLE，或新库重跑
事务没回滚干净失败后仍有残留行闭包里是否误用了 db 而非 tx；关联写入是否走了 Association 的 tx 句柄

下周预告：第 4 周 JWT + 工程化

数据能存住了，但谁都能改任何数据——下周解决「这是谁在操作」。三件事：

  • 密码哈希：bcrypt，本周模型里那个 Password varchar(255) 就是为 60 字符的 bcrypt 摘要预留的；注册时哈希、登录时 CompareHashAndPassword。
  • JWT 双 token：短期 access（15 分钟，随请求头带）+ 长期 refresh（7 天，只用来调 POST /auth/refresh 换新的 access）。为什么要两个？泄露半径：access 被偷了 15 分钟就失效，refresh 存得严严实实。第 6 周 React 的 axios 拦截器就靠这条链路实现「无感刷新」。
  • 工程化：.env + godotenv 把本周硬编码的 DSN 挪出去（这周的 const DSN 就是下周的配置项）、slog 结构化日志与请求 ID、优雅关闭、httptest 接口测试——本周换掉的内存版 repository 那时会变成测试桩，分层的好处再来一次。
