第 3 周总结：把 API 接到真实的数据库上

模型定义全文（含关联）

internal/model/user.gopackage model

import gorm.io/gorm

// gorm.Model = ID + CreatedAt + UpdatedAt + DeletedAt（自带软删除）
type User struct {
    gorm.Model

    Username string `gorm:"type:varchar(32);uniqueIndex;not null" json:"username"` // 登录名（没有 name 列）
    Nickname string `gorm:"type:varchar(32)" json:"nickname"`                     // 显示名
    Email    string `gorm:"type:varchar(128);uniqueIndex" json:"email"`
    Password string `gorm:"type:varchar(72);not null" json:"-"`                // 72 = bcrypt 上限；json:"-" 永不输出
    Role     Role   `gorm:"column:role;type:tinyint;not null;default:0;index" json:"role"`

    ArticleCount int64     `gorm:"column:article_count;type:bigint;not null;default:0" json:"article_count"` // 冗余计数，day5 事务里维护
    Articles     []Article `gorm:"foreignKey:UserID" json:"articles,omitempty"`
}

// Role 自定义整型：0 普通用户，1 管理员（第 8 周正式讲权限，本周只存不判）
type Role int8

const (
    RoleUser  Role = iota
    RoleAdmin
)
internal/model/article.gopackage model

import gorm.io/gorm

type Article struct {
    gorm.Model

    Title      string `gorm:"type:varchar(120);not null" json:"title"`
    Content    string `gorm:"type:text" json:"content"`
    Summary    string `gorm:"type:varchar(255)" json:"summary"`     // 列表页摘要，第 6 周前端直接吃它
    CoverImage string `gorm:"type:varchar(255)" json:"cover_image"` // 封面 URL，第 8 周上传复用同一列
    Views      int64  `gorm:"default:0" json:"views"`

    UserID uint  `gorm:"index" json:"user_id"`                    // 一对多外键：真正进表的列
    User   User  `gorm:"foreignKey:UserID" json:"user,omitempty"` // belongs to：不落列，Preload/Joins 才有值（键叫 user）
    Tags   []Tag `gorm:"many2many:article_tags" json:"tags,omitempty"` // 多对多：关系在中间表 article_tags
}
internal/model/tag.gopackage model

import time

// Tag 不用 gorm.Model：标签没有「改」也没有「软删」的需求 → db.Delete 发的是真 DELETE
type Tag struct {
    ID        uint      `gorm:"primaryKey" json:"id"`
    Name      string    `gorm:"type:varchar(32);not null;uniqueIndex" json:"name"`
    CreatedAt time.Time `json:"created_at"`
}

func (Tag) TableName() string { return "tags" }
cmd/server/main.go（装配 · 本周终态，完整可跑）package main

import (
    log
    net/http

    "github.com/gin-gonic/gin"

    "goblog/internal/config"
    "goblog/internal/handler"
    "goblog/internal/middleware"
    "goblog/internal/model"
    "goblog/internal/repository"
    "goblog/internal/service"
    "goblog/pkg/response"
)

func main() {
    db := repository.MustConnect(config.DSN)   // day1 的 db.go：gorm.Open + 连接池参数 + Ping，别再自己 gorm.Open
    if err := db.AutoMigrate(&model.User{}, &model.Article{}, &model.Tag{}); err != nil {
        log.Fatalf("迁移失败: %v", err)         // 中间表 article_tags 由 many2many 自动建
    }

    // 仓储：四个构造函数都返回接口类型，换实现只动这一段
    userRepo := repository.NewUserRepo(db)     // ← 上周这里是 NewMemRepo()，只改这一行
    artRepo := repository.NewArticleRepo(db)
    tagRepo := repository.NewTagRepo(db)
    statsRepo := repository.NewStatsRepo(db)    // day5 新增：维护 users.article_count

    // service：注入仓储。文章 service day5 起还要 db 开事务（3 参 → 5 参，day3 预告过）
    userSvc := service.NewUserService(userRepo)
    artSvc := service.NewArticleService(artRepo, userRepo, tagRepo, statsRepo, db)

    r := gin.Default()
    r.Use(middleware.CORS(), middleware.Timer())
    // 健康检查：无前缀，但仍走统一 Body（response.OK 包成 {code:0,message,data}）
    r.GET("/health", func(c *gin.Context) { response.OK(c, gin.H{"status": "ok"}) })

    api := r.Group("/api/v1")                  // 读接口挂这里
    auth := r.Group("/api/v1")                 // 同前缀再开一个带鉴权的组
    auth.Use(middleware.RequireAuth())          // Bearer go-fullstack-2026，第 4 周换 JWT
    handler.NewUserHandler(userSvc).Register(api, auth)
    handler.NewArticleHandler(artSvc).Register(api, auth)

    r.NoRoute(func(c *gin.Context) {   // 兜底：未匹配路由 → 404 + 通用码 1001
        response.Fail(c, http.StatusNotFound, response.CodeUserNotFound, "接口不存在: "+c.Request.URL.Path)
    })
    log.Fatal(r.Run(":8080"))
}

从内存版改造成 MySQL 版的步骤（迁移清单）

#改造动作注意点
1模型从 struct + 手写 ID 自增 换成带 tag 的 GORM 模型（internal/model）给每个字段补 json tag，密码 json:"-"，否则响应结构一变前端就崩
2repository 接口保持不变，新增 GORM 实现（NewUserRepo(db) UserRepository、NewArticleRepo、NewTagRepo、day5 的 NewStatsRepo）构造函数返回接口类型，编译器立刻能查出你漏实现了哪个方法
3把「内存里用 for 循环过滤」的查询换成 Where；「切片追加」换成 Createservice 层默认不出现 *gorm.DB；唯一例外是 day5 给 ArticleService 注入的 db，它只用来 Transaction 开事务，绝不拿它直接 Where/Find
4「找不到就返回 nil, nil」的语义改成 ErrXxxNotFound + errors.Is让 handler 能区分 404 与 500，这是本周错误口径的关键升级；nil, nil 从此绝迹
5删掉 map + RWMutex 的并发保护代码内存版才需要自己加锁；并发正确性交给数据库与事务（D5）
6main.go 里只换装配那几行（repo 构造函数 + 路由组）内存版留在 gin-api 模块当测试桩参考，不搬进 goblog（day1 承接清单第 7 行），所以给接口加方法时不会撞到旧实现

做完这六步你会直观地看到：repository 和 model 整个换掉、文章层（service/handler/dto）本周从零新增，而用户层 service/handler 几乎零改动——换 GORM 实现没逼它们改一行。这就是第 2 周分层与第 1 周接口设计的回报——依赖倒置不是术语，是「换实现不重写」这件事本身。

列表接口 800ms日志几十条 SELECTN+1 → 改 Preload；排序列加索引；SELECT * 换成需要的列
AutoMigrate 不生效改了 tag 表结构没变它不删列不改类型；手工 ALTER TABLE，或新库重跑
事务没回滚干净失败后仍有残留行闭包里是否误用了 db 而非 tx；关联写入是否走了 Association 的 tx 句柄

下周预告：第 4 周 JWT + 工程化

数据能存住了，但谁都能改任何数据——下周解决「这是谁在操作」。第 4 周不另起项目，就在 goblog 这个模块里原地加 auth/config/logger（W4 是 W3 的延续，模块名不换、目录不推倒重来）。三件事：

  • 密码哈希：bcrypt，本周模型里那个 Password varchar(72)（day2 已定）就是为 60 字符的 bcrypt 摘要预留的；注册时哈希、登录时 CompareHashAndPassword。
  • JWT 双 token：短期 access（15 分钟，随请求头带）+ 长期 refresh（7 天，只用来调 POST /api/v1/auth/refresh 换新的 access）。两个密钥分开：JWT_SECRET 签 access、JWT_REFRESH_SECRET 签 refresh。为什么要两个？泄露半径：access 被偷了 15 分钟就失效，refresh 存得严严实实。第 6 周 React 的 axios 拦截器就靠这条链路实现「无感刷新」。
  • 工程化：.env + godotenv 把本周硬编码的 const DSN 挪出去——config 读 DB_HOST / DB_PORT / DB_USER / DB_PASS / DB_NAME（外加 PORT / GIN_MODE / JWT_SECRET / JWT_REFRESH_SECRET）再拼 DSN，库名/账号仍是那套 goblog；再加 slog 结构化日志与请求 ID、优雅关闭、httptest 接口测试——本周留在 gin-api 的内存版 repository 那时会变成测试桩，分层的好处再来一次。

提前做准备：本周项目里给 Article 留好 UserID（你已经有了），下周的「仅作者可编辑/删除」就直接依赖它；本周 POST /api/v1/articles 已经挂了 middleware.RequireAuth()（写死的假 Bearer go-fullstack-2026），第 4 周把它整段换成验 JWT 的真 middleware.Auth(secret)——届时未登录才是 401 + code:1005。进入第 4 周前，先确保本周的 8 条验收全绿。
