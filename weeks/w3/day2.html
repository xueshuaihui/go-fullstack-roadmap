知识讲义

1. 模型定义：gorm.Model 与字段 tag

gorm.Model 是一个官方内嵌结构体，塞进来就有 ID / CreatedAt / UpdatedAt / DeletedAt 四个字段（含软删除）：

gorm.Model 展开type Model struct {
    ID        uint           `gorm:"primaryKey"`
    CreatedAt time.Time
    UpdatedAt time.Time
    DeletedAt gorm.DeletedAt  `gorm:"index"`  // ← 有这个字段就自动软删除，Day 5 细讲
}

本周主线模型的完整定义（三个文件一起给出，后三天直接复用）：

internal/model/user.gopackage model

import gorm.io/gorm

type User struct {
    gorm.Model

    // column: 列名 | type: 建表类型 | size: 长度 | not null: 非空 | uniqueIndex: 唯一索引
    Name     string `gorm:"column:name;type:varchar(64);size:64;not null" json:"name"`
    Email    string `gorm:"type:varchar(128);size:128;not null;uniqueIndex" json:"email"`
    Password string `gorm:"type:varchar(255);not null" json:"-"` // json:"-" 永不输出到响应

    Articles []Article `gorm:"foreignKey:UserID" json:"articles,omitempty"` // Day 3 用
}
internal/model/article.gopackage model

import gorm.io/gorm

type Article struct {
    gorm.Model

    Title   string `gorm:"type:varchar(128);size:128;not null;index" json:"title"`
    Content string `gorm:"type:text" json:"content"`
    Views   int    `gorm:"not null;default:0" json:"views"`

    UserID uint  `gorm:"type:bigint unsigned;not null;index" json:"user_id"` // 一对多外键
    User   User  `json:"user,omitempty"`                                    // belongs to
    Tags   []Tag `gorm:"many2many:article_tags;" json:"tags,omitempty"`       // 多对多中间表
}
internal/model/tag.gopackage model

import (
    time

    gorm.io/gorm
)

type Tag struct {
    ID        uint      `gorm:"primaryKey" json:"id"`       // 自定义：不用 gorm.Model
    Name      string    `gorm:"type:varchar(32);size:32;not null;uniqueIndex" json:"name"`
    CreatedAt time.Time `json:"created_at"`                 // 只要创建时间，无更新/软删
}

// 表名单数化：GORM 默认把结构体名转 snake_case + 复数（Tag → tags 已经对了）
func (Tag) TableName() string { return "tags" }
tag作用备注
primaryKey指定主键字段名 ID 或 XxxID 会被自动认成主键/外键，所以自定义主键要显式写
column:"x"指定列名不写则用 snake_case（UserID → user_id）
type:"varchar(64)"建表用的数据库类型只在 AutoMigrate 建表/首次加列时生效
size:64GORM 自己的长度声明（varchar(64)）和 type 二选一即可，别写矛盾的两个值
not null / default:0非空 / 默认值字符串默认值要转义双引号：default:'' 或直接 default:"x"
index / uniqueIndex普通/唯一索引可带名字与组合：uniqueIndex:idx_user_name 出现在多个字段上即联合索引
autoIncrement:false自定义主键（如 UUID / slug）本周不用，知道即可
json:"-" / json:"omitempty"不是 GORM 的，是 encoding/json 的密码永不回前端；关联字段不请求就不返回

命名规则默认值：结构体 ArticleTag → 表 article_tags；字段 UserID → 列 user_id。全项目要么遵守默认，要么统一用 TableName()，别混着来。

2. Create：插入并回填自增主键

repository/crud_demo.gofunc demo(db *gorm.DB) {
    u := model.User{Name: "小明", Email: "xm@example.com", Password: "hashed"}
    if err := db.Create(&u).Error; err != nil { /* 唯一索引冲突等 */ }
    fmt.Println(u.ID)        // ← 1：插入后 ID / CreatedAt 自动回填到 struct，这是 Create 的关键福利

    if err := db.Create(&model.User{Name: "阿花", Email: "ah@example.com", Password: "h"}).Error; err != nil {
        log.Println(err)
    }
}

唯一索引冲突在 MySQL 驱动下会返回 Error 1062 (23000): Duplicate entry 'xm@example.com' for key 'users.email'。注册接口要靠它兜底（先查再插存在并发窗口，Day 4/第 4 周会强调），今天先认识这个错误串。

3. 查询四件套：First / Take / Last / Find

方法生成的 SQL 关键部分找不到时用途
db.First(&u, 1)ORDER BY id LIMIT 1ErrRecordNotFound按主键/条件取「最早的一条」，最常用
db.Take(&u, 1)LIMIT 1（无排序）ErrRecordNotFound不关心顺序、想让 MySQL 自己挑最优时
db.Last(&u)ORDER BY id DESC LIMIT 1ErrRecordNotFound取最新一条
db.Find(&users)SELECT * FROM users（无 LIMIT）无错（得到空 slice）取多条，列表接口用它
db.Find(&users, 1)WHERE id = 1 仍返回 slice无错（空 slice）按主键批量
查询写法// ① 传结构体做条件：非零字段才进 WHERE（零值 0 / "" / nil 被忽略！）
cond := model.User{Name: "小明"}          // Email 是 "" → 不参与条件
db.Where(&cond).Find(&users)              // WHERE name = '小明'

// ② 传 map：键是列名，零值也算条件（这是查「Views = 0」的唯一稳妥方式）
db.Where(map[string]any{"name": "小明", "views": 0}).Find(&articles)

// ③ 链式 + 占位符（最常用，占位符 ? 由驱动转义，天然防注入）
db.Where("name = ? AND email LIKE ?", "小明", "%example%").
   Order("id desc").
   Limit(5).
   Find(&users)

// 取部分列 / 标量
db.Select("id", "name").Find(&users)
var emails []string
db.Model(&model.User{}).Where("id < ?", 10).Pluck("email", &emails)

4. 更新：Save vs Updates，以及零值陷阱

更新对比// Save：主键非零 → UPDATE 全部列（等于把整个 struct 覆盖过去）
var u model.User
db.First(&u, 1)
u.Name = "小明改名"
db.Save(&u)      // UPDATE users SET name=...,email=...,password=...,created_at=... WHERE id=1
                 // 主键为 0 时 Save 会退化成 INSERT —— 新人最常见的意外

// Updates + struct：只更新非零字段（Views=0 会被悄悄跳过！）
db.Model(&model.User{}).Where("id = ?", 1).Updates(model.User{Name: "批量改", Email: "b@x.com"})

// Updates + map：零值照常写入 —— 归零/清空必须用它
db.Model(&a).Updates(map[string]any{"views": 0, "content": ""})

// 单列 / 表达式自增（DB 端计算，避免读改写并发丢失）
db.Model(&model.Article{}).Where("id = ?", 1).Update("title", "新标题")
db.Model(&model.Article{}).Where("id = ?", 1).UpdateColumn("views", gorm.Expr("views + ?", 1))

// 影响行数：更新 0 行不是错误，RowsAffected 才是唯一线索
res := db.Model(&model.User{}).Where("id = ?", 99999).Update("name", "x")
if res.Error != nil { return res.Error }
if res.RowsAffected == 0 { return errNoUser }   // 否则你会以为改成功了

对比 JS：Object.assign(user, patch) 里 undefined 不覆盖、null 覆盖。GORM 的 struct 版 Updates 是「非零值才覆盖」——Go 没有 undefined，只有零值，所以想表达「把它置空」必须换 map（或用 *int 这类指针字段，第 4 周讲）。

5. 删除与「防止一句删全表」

删除db.Delete(&model.User{}, 1)                       // 有 DeletedAt → UPDATE deleted_at；无 → DELETE
db.Delete(&model.Article{}, []uint{1, 2, 3})      // 批量按主键
db.Where("user_id = ?", 1).Delete(&model.Article{}) // 条件删除
res := db.Where("id = ?", 99).Delete(&model.User{})
fmt.Println(res.RowsAffected)                       // 0 表示其实没删到（软删除也一样）

GORM 有个隐藏保险：全局配置开 BlockGlobalUpdate: true 后，不带 Where 的 Update/Delete 会直接报 ErrMissingWhereClause，避免你把全表改了。项目里推荐打开：

db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{
    SkipDefaultTransaction: false,
    BlockGlobalUpdate:      true,   // ← 没有 WHERE 的 UPDATE/DELETE 直接报错
})

6. ErrRecordNotFound：出错还是没数据？

repository/user.goimport (
    errors

    gorm.io/gorm
)

var ErrUserNotFound = errors.New("用户不存在")

func (r *gormUserRepo) ByID(id uint) (*model.User, error) {
    var u model.User
    err := r.db.First(&u, id).Error
    switch {
    case errors.Is(err, gorm.ErrRecordNotFound):
        return nil, ErrUserNotFound      // → handler 返回 404 + code 1001
    case err != nil:
        return nil, err                  // → 500，数据库真出问题了，别混成 404
    default:
        return &u, nil
    }
}

如果你写过 Sequelize，它的 findByPk 找不到时默认 throw；GORM 不 panic 也不 throw，而是把 ErrRecordNotFound 放在 db.Error 里。必须用 errors.Is 而不是 ==——将来错误被 fmt.Errorf("%w", ...) 包装过，== 就比不上了。

练习 2：把内存 repository 换成 GORM（主线，90 分钟）

保持第 2 周定义好的接口不动，新增 GORM 实现并在 main.go 换装配。要求 POST /users、GET /users、GET /users/:id、PUT /users/:id、DELETE /users/:id 全部行为与上周一致，只是数据落在 MySQL。

internal/repository/user.gopackage repository

import (
    errors

    "goblog/internal/model"
    gorm.io/gorm
)

// 接口：第 2 周就是它，今天一个字都不改
type UserRepo interface {
    Create(u *model.User) error
    ByID(id uint) (*model.User, error)
    ByEmail(email string) (*model.User, error)
    All() ([]model.User, error)
    Update(u *model.User) error
    Delete(id uint) error
}

type gormUserRepo struct { db *gorm.DB }

func NewGormUserRepo(db *gorm.DB) UserRepo { return &gormUserRepo{db: db} }

func (r *gormUserRepo) Create(u *model.User) error { return r.db.Create(u).Error }

func (r *gormUserRepo) ByID(id uint) (*model.User, error) {
    var u model.User
    if err := r.db.First(&u, id).Error; err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) { return nil, ErrUserNotFound }
        return nil, err
    }
    return &u, nil
}

func (r *gormUserRepo) ByEmail(email string) (*model.User, error) {
    var u model.User
    err := r.db.Where("email = ?", email).Take(&u).Error
    if errors.Is(err, gorm.ErrRecordNotFound) { return nil, nil } // 「没查到」是正常的 nil
    if err != nil { return nil, err }
    return &u, nil
}

func (r *gormUserRepo) All() ([]model.User, error) {
    users := []model.User{}                       // 预置空 slice，保证 JSON 输出 [] 而不是 null
    err := r.db.Order("id asc").Find(&users).Error
    return users, err
}

func (r *gormUserRepo) Update(u *model.User) error { return r.db.Save(u).Error }

func (r *gormUserRepo) Delete(id uint) error {
    res := r.db.Delete(&model.User{}, id)
    if res.Error != nil { return res.Error }
    if res.RowsAffected == 0 { return ErrUserNotFound }
    return nil
}

在 handler 里把 :id 转成 uint 交给 repository，非法 id 返回 400，找不到返回 404 + code:1001。

internal/handler/user.goid, err := strconv.ParseUint(c.Param("id"), 10, 64)
if err != nil || id == 0 {
    c.JSON(400, gin.H{"code": 1004, "message": "id 必须是正整数", "data": nil})

u, err := h.svc.Get(uint(id))

常见坑（新手 100% 会踩）

坑现象正确姿势
把 db 当一次性对象上一条链式的 Where 神秘地影响了下一条查询每次查询从 r.db 起链；若复用需 db.Session(&gorm.Session{})
Updates 传 struct 想归零SQL 里根本没有该列，值没变用 map[string]any{...}，或指针字段
Save 传了零 ID本意更新，结果插了一条新行先 First 拿到带 ID 的对象再 Save，或用 Updates
不判 RowsAffected更新/删除了不存在的记录却返回 200 successRowsAffected == 0 → 404
用 err == gorm.ErrRecordNotFound错误被包装后判断失效，500 满天飞errors.Is(err, gorm.ErrRecordNotFound)
忘写 json tag响应里出现 DeletedAt、Password、字段名变 PascalCase模型上写清 json:"xxx"，密码用 json:"-"
Where 传结构体用了零值想查 views=0 却查出全部零值条件用 map 或字符串占位
AutoMigrate 后改类型tag 改了 varchar(64)→255，库里没变手工 ALTER TABLE，或第 7 周版本化迁移
