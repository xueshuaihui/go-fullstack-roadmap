知识讲义

1. 已读的边界：只读自己的

UPDATE 的 WHERE 一次带齐四个条件：

片段 · 已读更新的完整条件UPDATE homeos_notification
   SET read_at = now()
 WHERE family_id = ?          -- 家庭隔离，scope 铁律
   AND member_id = ?          -- 只能替自己已读：别人的红点轮不到你消
   AND read_at IS NULL        -- 已读的不再动，天然幂等
   AND id IN (?)              -- 或 all=true 时省略本行

妙处在 member_id 条件的双重身份：它既是权限判定，又是数据过滤。别人家的、别人收件箱里的 id 混进 ids，这条 UPDATE 命中 0 行——静默跳过，不报错。为什么不报 1001？因为「这个 id 存不存在」本身就是不该泄露的信息（S4 起的老纪律：越权与不存在同形）。重复提交同一批 ids 也无副作用：read_at IS NULL 挡住了第二次写。

2. 第三个出口：read 的响应也带 unread

已读操作成功后，客户端最想要的下一个数字就是「还剩几条未读」。让它再发一次 GET 是浪费——POST read 的响应直接带同一份 unread（还是调 repo.UnreadCount，一个字都不许多写）。至此三个出口齐了：summary、GET notifications、POST read。今天的测试把昨天的双出口断言升级成三口同值：已读 2 条后，三个出口一起从 3 变 1。

3. push-token：只登记，不承诺送达

POST /api/homeos/devices/push-token 做的事很薄：客户端把「这台设备（device_id）、什么平台（platform）、通道 token 是什么」报上来，服务端按 (member_id, device_id) upsert 进 homeos_device。同一台设备换 token（重装、换通道账号）是更新不是新增；同一个人的第二台设备是新增一行。真正的推送下发链路（对接厂商通道、失败重试）不在本课程范围——今天登记的数据就是将来那条链路的原料，这里只作前瞻，不承诺周次。

步骤 2 · service：已读与设备登记

片段 · server/internal/homeos/service/notification.go 追加// MarkRead 单条或全部已读；返回受影响行数。只动 member_id=me 且未读的行。
func (s *NotificationService) MarkRead(ctx context.Context, familyID, memberID int64, ids []int64, all bool) (int64, error) {
    q := s.db.WithContext(ctx).Model(&model.Notification{}).
        Where("family_id = ? AND member_id = ? AND read_at IS NULL", familyID, memberID)
    if !all {
        if len(ids) == 0 { return 0, response.ErrInvalidParam } // 两种形态都没有：400 + 1004
        q = q.Where("id IN ?", ids)
    }
    res := q.Updates(map[string]any{"read_at": gorm.Expr("now()"), "updated_at": gorm.Expr("now()")})
    return res.RowsAffected, res.Error
}

// UpsertDevice 按 (member_id, device_id) upsert：换 token 是更新，换设备是新增。
func (s *NotificationService) UpsertDevice(ctx context.Context, familyID, memberID int64, deviceID, platform, token string) error {
    if deviceID == "" || token == "" { return response.ErrInvalidParam }
    d := model.Device{FamilyID: familyID, MemberID: memberID, DeviceID: deviceID, Platform: platform, PushToken: token, Version: 1}
    return s.db.WithContext(ctx).
        Clauses(clause.OnConflict{
            Columns:   []clause.Column{{Name: "member_id"}, {Name: "device_id"}},
            DoUpdates: clause.Assignments(map[string]any{"push_token": token, "platform": platform, "updated_at": gorm.Expr("now()")}),
        }).
        Create(&d).Error
}

步骤 3 · handler：第三个出口

片段 · server/internal/homeos/handler/notification.go 追加// Read POST /api/homeos/notifications/read
// body: {"ids":[1,2]} 或 {"all":true}；响应 data 带受影响行数与最新 unread。
func (h *NotificationHandler) Read(c *gin.Context) {
    familyID := c.GetInt64("familyID")
    memberID := c.GetInt64("memberID")
    var req struct {
        IDs []int64 `json:"ids"`
        All bool    `json:"all"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, http.StatusBadRequest, response.CodeInvalidParam, "参数错误")
        return
    }
    n, err := h.svc.MarkRead(c.Request.Context(), familyID, memberID, req.IDs, req.All)
    if err != nil { response.FailErr(c, err); return }

    unread, err := h.svc.Unread(c.Request.Context(), familyID, memberID) // 还是那一个函数
    if err != nil { response.FailErr(c, err); return }
    response.OK(c, gin.H{"marked": n, "unread": unread})
}

// PushToken POST /api/homeos/devices/push-token
func (h *NotificationHandler) PushToken(c *gin.Context) {
    var req struct {
        DeviceID  string `json:"device_id" binding:"required"`
        Platform  string `json:"platform" binding:"required"`
        PushToken string `json:"push_token" binding:"required"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, http.StatusBadRequest, response.CodeInvalidParam, "参数错误")
        return
    }
    err := h.svc.UpsertDevice(c.Request.Context(), c.GetInt64("familyID"), c.GetInt64("memberID"),
        req.DeviceID, req.Platform, req.PushToken)
    if err != nil { response.FailErr(c, err); return }
    response.OK(c, gin.H{"registered": true})
}

// 路由：homeos.POST("/notifications/read", nh.Read)
//       homeos.POST("/devices/push-token", nh.PushToken)

常见坑（新手 100% 会踩）

坑现象正确姿势
WHERE 少了 read_at IS NULL重复点「全部已读」时 updated_at 被反复刷，乐观锁版本白白上涨只更新未读行，天然幂等
把别人的 id 报 1001泄露了「这个 id 存在」——越权与存在性同形的纪律破了静默不命中：marked 少一个数而已
read 响应不带 unread前端只好再发一次 GET，红点闪一下才归零第三出口就在响应里；还是调那唯一的 count 函数
{} 空 body 当全部已读误触一次全清，用户炸毛ids 与 all 两者皆空 → 400 + 1004；「全部」必须是显式意图
push-token 每次 INSERT一台设备半年攒 200 行，将来推送发 200 遍(member_id, device_id) 唯一 + upsert
device 表用 user_id 不用 member_id同一账号在两个家庭里的设备语境混在一起通知域的钥匙统一是 member_id（当前会话家庭里的成员）

术语一句话解释
已读回写read_at 落 now()；条件四件套 family/member/未读/ids
静默不命中越权 id 混入时命中 0 行、不报错——存在性不泄露

(member_id, device_id)设备行的幂等键：同人同机一行，换 token 是更新
