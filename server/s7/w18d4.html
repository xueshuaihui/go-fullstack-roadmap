动态流接口：按天分组、按面筛选、游标分页

动态流是首页 D 区的内容源：「家里刚发生了什么」——谁记了一笔账、谁改了面配置、谁的待办完成了。今天建 homeos_dynamic 表与 GET /api/homeos/dynamics 接口，三个能力一次做齐：按天分组（家庭时区的天）、按面筛选（code 参数）、游标分页（不透明游标，续读不重不漏）。首页 summary 的 D 区取前 20 条，与动态流页续读同一个接口、同一个查询函数——同源纪律第三次上岗。

今日目标

知识讲义

1. 动态行长什么样

一行 = 家里发生的一件事。关键列的取舍：

  • module_code：来源面（finance / homeos / …）。「按面筛选」直接过滤这一列；它也决定了深链回源对象的路由域。
  • actor_member_id：谁干的。渲染时联出名字与称谓——注意这是进程内同库查询，不是跨面：动态表与成员表都在 homeos。
  • kind + summary：机读分型 + 人读文案。文案在写入时定稿（「妈妈记了一笔 34.00 的餐饮支出」），读侧不做模板拼接——否则改一次文案样式，历史动态全变脸。
  • occurred_at：事情发生的时刻（TIMESTAMPTZ），不是行创建时刻。排序与游标都用它。
  • 没有 source_id 的外键：动态引用源对象走逻辑 id，不建物理外键、不复制金额——源对象被删，动态行照常在（历史不被追溯篡改）。

写入侧的纪律顺带立好：别的模块（如 finance 记账动作）要发动态，不直插这张表，走 platform 的追加接口（与 FamilyDirectory、DueRegistry 同一条跨模块纪律）。今天先把表与读侧做完，写入用测试造数；追加接口在收口日前补一个最小版（见步骤 4）。

2. 按天分组：SQL 管序，Go 管组

「天」是谁的天？家庭时区的天（S7·D8 立的口径：today 按家庭时区，动态分组同理）。实现上两种做法：

做法评价
SQL 里 GROUP BY date(occurred_at AT TIME ZONE ?)能跑，但游标分页与分组在 SQL 里纠缠：页边界切在一天的中间时，「组」跨页了，拼接逻辑非常绕
SQL 只负责排序 + 游标 + LIMIT，分组在 Go 侧扫一遍本页内顺序折叠成 []DayGroup；跨页的同一天由客户端按 date 合并（前端侧的续读契约）。SQL 简单、可测、口径集中

游标照 S3 的口径：不透明，服务端把 (occurred_at, id) 编码进 base64；客户端不得解析、不得自造。排序键必须唯一化——同一微秒可能有两条动态，只按 occurred_at 排会翻页重复，所以游标带上 id 做决胜键（tie-breaker）。解不开/解出畸形值 → 400 + 1004（参数错误），不静默当第一页。

3. summary 与 dynamics：一个查询函数两个出口

结构Query(familyID, code, cursor, limit)
   ├─ 出口①  summary.dynamics.items：code=""、cursor=""、limit=20 → 平铺 items + has_more
   └─ 出口②  GET /api/homeos/dynamics：带筛选与游标 → 分组 []DayGroup + next_cursor + has_more

分组只是出口②的呈现整形，查询内核一份。首页 D 区取前 20 条、动态流页续读同一接口——用户从首页滑进动态流页，看到的前 20 条与首页完全一致，续读从第 21 条开始，不重不漏。

import (
    "context"
    "encoding/base64"
    "fmt"
    "strconv"
    "strings"
    "time"

    "gorm.io/gorm"

    "homecube/internal/homeos/model"
    "homecube/pkg/response"
)

type Dynamic struct {
    ID            int64     `gorm:"primaryKey" json:"id"`
    FamilyID      int64     `gorm:"not null" json:"-"`
    ModuleCode    string    `gorm:"not null" json:"code"`
    Kind          string    `gorm:"not null" json:"kind"`
    ActorMemberID int64     `gorm:"not null" json:"-"`
    Summary       string    `gorm:"not null" json:"summary"`
    SourceID      *int64    `json:"source_id"`
    OccurredAt    time.Time `gorm:"not null" json:"occurred_at"`
    CreatedAt     time.Time `json:"-"`
    UpdatedAt     time.Time `json:"-"`
    DeletedAt     *time.Time `json:"-"`
    DeletedBy     *int64    `json:"-"`
    Version       int       `json:"-"`

    ActorName     string `gorm:"-" json:"actor_name"`     // 联出的展示字段
    ActorRelation string `gorm:"-" json:"actor_relation"`
}

func (Dynamic) TableName() string { return "homeos_dynamic" }

// EncodeCursor 不透明游标：base64(微秒|id)。客户端不得解析。
func EncodeCursor(occurredAt time.Time, id int64) string {
    raw := fmt.Sprintf("%d|%d", occurredAt.UnixMicro(), id)
    return base64.RawURLEncoding.EncodeToString([]byte(raw))
}

// DecodeCursor 解不开或畸形 → response.ErrInvalidParam（400 + 1004），不静默当第一页。
func DecodeCursor(s string) (time.Time, int64, error) {
    b, err := base64.RawURLEncoding.DecodeString(s)
    if err != nil {
        return time.Time{}, 0, response.ErrInvalidParam
    }
    ts, idStr, ok := strings.Cut(string(b), "|")
    if !ok {
        return time.Time{}, 0, response.ErrInvalidParam
    }
    micro, err1 := strconv.ParseInt(ts, 10, 64)
    id, err2 := strconv.ParseInt(idStr, 10, 64)
    if err1 != nil || err2 != nil {
        return time.Time{}, 0, response.ErrInvalidParam
    }
    return time.UnixMicro(micro), id, nil
}

// Query 是动态流的唯一查询内核，summary D 区与 dynamics 接口共用。
// code 为空 = 不按面筛选；cursor 为空 = 第一页。多取一条判 has_more。
func Query(ctx context.Context, db *gorm.DB, familyID int64, code, cursor string, limit int) ([]Dynamic, bool, string, error) {
    q := db.WithContext(ctx).Model(&Dynamic{}).
        Where("family_id = ?", familyID)
    if code != "" {
        q = q.Where("module_code = ?", code)
    }
    if cursor != "" {
        at, id, err := DecodeCursor(cursor)
        if err != nil {
            return nil, false, "", err
        }
        // (occurred_at, id) 双键严格小于：同刻两条也不重不漏
        q = q.Where("(occurred_at, id) < (?, ?)", at, id)
    }
    var rows []Dynamic
    if err := q.Order("occurred_at DESC, id DESC").
        Limit(limit + 1).Find(&rows).Error; err != nil {
        return nil, false, "", err
    }
    hasMore := len(rows) > limit
    if hasMore {
        rows = rows[:limit]
    }
    next := ""
    if hasMore && len(rows) > 0 {
        last := rows[len(rows)-1]
        next = EncodeCursor(last.OccurredAt, last.ID)
    }
    if err := fillActors(ctx, db, familyID, rows); err != nil {
        return nil, false, "", err
    }
    return rows, hasMore, next, nil
}

// fillActors 批量联出名字与称谓：一次 IN 查询，禁止每行查一次（N+1）。
func fillActors(ctx context.Context, db *gorm.DB, familyID int64, rows []Dynamic) error {
    ids := map[int64]bool{}
    for _, r := range rows {
        ids[r.ActorMemberID] = true
    }
    idList := make([]int64, 0, len(ids))
    for id := range ids {
        idList = append(idList, id)
    }
    if len(idList) == 0 {
        return nil
    }
    var members []model.Member
    if err := db.WithContext(ctx).
        Where("family_id = ? AND id IN ?", familyID, idList).
        Find(&members).Error; err != nil {
        return err
    }
    byID := map[int64]model.Member{}
    for _, m := range members {
        byID[m.ID] = m
    }
    for i := range rows {
        if m, ok := byID[rows[i].ActorMemberID]; ok {
            rows[i].ActorName = m.Name
            rows[i].ActorRelation = m.Relation
        }
    }
    return nil
}

步骤 3 · dynamics handler + summary D 区换真

import (
    "strconv"
    "time"

    "github.com/gin-gonic/gin"
    "gorm.io/gorm"

    "homecube/pkg/response"
)

// DayGroup 是按天分组后的一天的形状。
type DayGroup struct {
    Date  string    `json:"date"` // 家庭时区的 YYYY-MM-DD
    Items []Dynamic `json:"items"`
}

// List GET /api/homeos/dynamics?code=finance&cursor=…&limit=20
func List(db *gorm.DB, tzOf func(familyID int64) *time.Location) gin.HandlerFunc {
    return func(c *gin.Context) {
        familyID := c.GetInt64("familyID")
        limit := 20
        if s := c.Query("limit"); s != "" {
            n, err := strconv.Atoi(s)
            if err != nil || n < 1 || n > 50 {
                response.Fail(c, 400, response.CodeInvalidParam, "limit 取 1-50")
                return
            }
            limit = n
        }
        rows, hasMore, next, err := Query(c.Request.Context(), db,
            familyID, c.Query("code"), c.Query("cursor"), limit)
        if err != nil {
            response.FailErr(c, err)
            return
        }

        // 分组在 Go 侧：行已按时间倒序，顺序折叠即可
        tz := tzOf(familyID)
        groups := []DayGroup{}
        for _, r := range rows {
            day := r.OccurredAt.In(tz).Format("2006-01-02")
            if len(groups) == 0 || groups[len(groups)-1].Date != day {
                groups = append(groups, DayGroup{Date: day})
            }
            g := &groups[len(groups)-1]
            g.Items = append(g.Items, r)
        }
        response.OK(c, gin.H{"list": groups, "next_cursor": next, "has_more": hasMore})
    }
}

// DynamicInput 是一条待追加的动态。跨模块写入只准走本接口，

    ActorMemberID int64
    Summary       string
    SourceID      *int64

type DynamicAppender interface {
    Append(ctx context.Context, in DynamicInput) error
}

常见坑（新手 100% 会踩）

坑现象正确姿势
游标只用 occurred_at同一时刻两条动态，翻页时重复或漏掉一条排序键唯一化：(occurred_at, id) 双键，游标编码两者
畸形游标静默当第一页客户端拿着坏游标无限重刷第一页，还以为到底了解码失败 → 400 + 1004，让错误可见
客户端解析游标改参数服务端换排序键，老客户端全崩游标不透明是契约；客户端只存、只回传
每行动态查一次成员（N+1）20 行动态 = 21 条 SQL，首页越刷越慢fillActors 一次 IN 批量联
summary 里另写一份动态查询首页与动态流页前 20 条不一致同一 Query 内核，两个出口只差呈现整形
分组用服务器本地时区跨时区家庭的「今天」错位，凌晨动态归错天按家庭时区分组（与 today 口径同源）
finance 直插 homeos_dynamic跨模块直查/直写，拆服务时全断走 platform.DynamicAppender 接口
has_more 用「本页满 limit」判断恰好整除时多翻一页空页多取一条：取 limit+1，多出即 has_more（今天的实现）

术语一句话解释
