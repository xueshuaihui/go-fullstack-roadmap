home/summary 骨架：faces[] 与面清单同源

本周做首页。壳基座对首页的要求只有一条硬指标：首屏只有这一个业务请求。GET /api/homeos/home/summary 一次聚合首页四个区（家庭区、今日区、服务矩阵、动态流）加未读数。今天先把骨架立起来，并且把服务矩阵 faces[] 做成真实现——它与上周的面清单接口同源合成：同一个 Compose 函数、同一份响应类型，契约测试断言两个接口逐项一致。其余三个区今天占位，D8-D10 逐个换真。

今日目标

知识讲义

1. 为什么首屏只许一个业务请求

移动端首屏每多一个请求，就多一次串行的 DNS/握手/鉴权/往返，弱网下白屏时间线性叠加。更隐蔽的代价是口径：如果首页的家庭信息、面矩阵、未读数分别来自三个接口，三个接口各有各的缓存与失败态，用户就会看到「矩阵里有财务、点进去说面未启用」这种自相矛盾的界面。

所以约定：客户端不得为渲染首页并行请求多个接口拼装。服务端把首屏需要的一切在一次响应里给全，且这些数据在服务端同一时刻、同一事务视图里算出来——天然自洽。代价是 summary 变重，这是刻意选择：重在一个可缓存、可优化的服务端聚合点，好过重在三万个客户端的串行请求链上。

2. faces[] 与 family/modules：同源不是相似，是同一个函数

两个接口都下发面清单，字段口径必须逐项一致：code/name/icon/born/availability/enabled/visible/version。「一致」的实现方式只有一种是对的：

做法评价
两个 handler 各查各的库、各拼各的 JSON错。今天一致，三个月后某一边加了个字段/改了个默认值，就漂移了
summary 内部 HTTP 调 family/modules 再转发错。进程内自己打自己，多一次序列化与鉴权，纯属行为艺术
两个 handler 都调 faces.Compose对。合成逻辑只有一份（S7·D3 立的规矩），响应类型也是同一个 faces.Item

一致性不靠自觉，靠契约测试钉住：同一个 token 分别打两个接口，断言 faces[] 与 list 逐项相等（七个字段全比）。比对口径里的两个重点：enabled ∧ visible 决定渲染（谓词上周写过）；owner 视角 visible 恒真。这个测试红了只有一种可能——有人在某一侧动了合成逻辑，而那正是它要拦的事故。

3. 骨架的纪律：占位要显式、要有替换日

今天的 summary 只有 faces[] 是真的，其余区先占位。占位不是随手塞个空对象完事，两条纪律：

  • 形状先定死：占位数据的 JSON 结构与最终一致（键名、嵌套、类型都对），只是值为零/空。客户端（前端线的壳基座卡）从第一天就能按最终契约写渲染。
  • 注释写明替换日：每个占位块标注「S7·D8 换真」「S7·D9 换真」。收口日（D12）的检查项之一就是全文件搜不到占位注释。

步骤 1 · 骨架 handler

完整 · server/internal/homeos/home/handler.gopackage home

import (
    "github.com/gin-gonic/gin"
    "gorm.io/gorm"

    "homecube/internal/homeos/faces"
    "homecube/internal/homeos/model"
    "homecube/pkg/response"
)

// Summary GET /api/homeos/home/summary —— 首屏唯一的业务请求。
func Summary(db *gorm.DB) gin.HandlerFunc {
    return func(c *gin.Context) {
        familyID := c.GetInt64("familyID")
        role := c.MustGet("role").(model.Role)

        // C 区 · 服务矩阵：与 GET family/modules 同源——同一个 Compose、同一个 Item 类型
        faceList, err := faces.Compose(c.Request.Context(), db, familyID, role)
        if err != nil {
            response.FailErr(c, err)
            return
        }

        data := gin.H{
            "faces": faceList,

            // A 区 · 家庭区：形状已定，S7·D8 换真
            "family": gin.H{
                "id": familyID, "name": "", "member_count": 0,
                "members": []gin.H{},
            },
            // B 区 · 今日区：形状已定，S7·D8 换真
            "today": gin.H{
                "due_today": gin.H{"count": 0, "items": []gin.H{}},
            },
            // D 区 · 动态流前 20 条：形状已定，S7·D10 换真
            "dynamics": gin.H{"items": []gin.H{}, "has_more": false},
            // 未读数：形状已定，S7·D9 换真
            "unread": gin.H{"count": 0},
        }
        response.OK(c, data)
    }
}
片段 · main.go 路由组（与 family/modules 同组，共用 S4 的 Auth 中间件）homeos.GET("/home/summary", home.Summary(db))

faces[] 用的是 faces.Item 原类型，不许在 home 包里再定义一个「长得一样」的 FaceVO。两个结构体字段相同不等于同源——序列化行为、日后的字段演进都会分叉。「同一份数据用同一个类型」是同源合成在代码层面的最后一道保险。

步骤 2 · 同源契约测试

完整 · server/internal/homeos/home/summary_contract_test.gopackage home

import (
    "encoding/json"
    "net/http/httptest"
    "testing"

    "github.com/gin-gonic/gin"

    "homecube/internal/homeos/faces"
    "homecube/internal/homeos/model"
    "homecube/internal/platform/testdb"
)

// decode 把统一响应信封的 data 解进目标。
func decode(t *testing.T, w *httptest.ResponseRecorder, into any) {
    t.Helper()
    var env struct {
        Code int             `json:"code"`
        Data json.RawMessage `json:"data"`
    }
    if err := json.Unmarshal(w.Body.Bytes(), &env); err != nil {
        t.Fatalf("信封解析失败: %v body=%s", err, w.Body.String())
    }
    if env.Code != 0 {
        t.Fatalf("code = %d, want 0", env.Code)
    }
    if err := json.Unmarshal(env.Data, into); err != nil {
        t.Fatalf("data 解析失败: %v", err)
    }
}

func TestFacesSameSourceAsModules(t *testing.T) {
    db := testdb.Open(t)
    famID, _ := faces.SeedFamilyForTest(t, db) // faces 包导出的 seed 助手（把 S7·D2 的 seedFamily 提为可导出即可）

    gin.SetMode(gin.TestMode)
    r := gin.New()
    // 不起真服务、不走 Auth：直接在路由上注入身份（S2 起的进程内测法）
    inject := func(role model.Role) gin.HandlerFunc {
        return func(c *gin.Context) {
            c.Set("familyID", famID)
            c.Set("memberID", int64(1))
            c.Set("role", role)
            c.Next()
        }
    }
    r.GET("/api/homeos/home/summary", inject(model.RoleOwner), Summary(db))
    r.GET("/api/homeos/family/modules", inject(model.RoleOwner), faces.ListModules(db))

    for _, role := range []model.Role{model.RoleOwner, model.RoleMember, model.RoleWard} {
        r2 := gin.New()
        r2.GET("/s", inject(role), Summary(db))
        r2.GET("/m", inject(role), faces.ListModules(db))

        ws := httptest.NewRecorder()
        r2.ServeHTTP(ws, httptest.NewRequest("GET", "/s", nil))
        wm := httptest.NewRecorder()
        r2.ServeHTTP(wm, httptest.NewRequest("GET", "/m", nil))

        var s struct {
            Faces []faces.Item `json:"faces"`
        }
        var m struct {
            List []faces.Item `json:"list"`
        }
        decode(t, ws, &s)
        decode(t, wm, &m)

        // 逐项一致：长度、顺序无关的集合级映射、七个字段全比
        if len(s.Faces) != len(m.List) {
            t.Fatalf("role=%d 长度不一致: %d vs %d", role, len(s.Faces), len(m.List))
        }
        byCode := map[string]faces.Item{}
        for _, it := range m.List {
            byCode[it.Code] = it
        }
        for _, f := range s.Faces {
            want, ok := byCode[f.Code]
            if !ok {
                t.Fatalf("role=%d summary 多发了 %s", role, f.Code)
            }
            if f != want { // Item 全是可比较字段，直接 == 逐项全比
                t.Fatalf("role=%d %s 不一致:\nsummary=%+v\nmodules=%+v", role, f.Code, f, want)
            }
            if role == model.RoleOwner && !f.Visible {
                t.Fatalf("owner 视角 %s visible 应恒真", f.Code)
            }
        }
    }
}

三个测试技巧值得记进 notes：① 信封解码用 json.RawMessage 二段式，data 形状各接口自定；② 不走 Auth 中间件，用一个注入身份的小中间件顶替——测的是 handler 契约，不是 S4 的鉴权；③ faces.Item 全部字段可比较，所以 f != want 一行就是「七个字段逐项一致」，日后 Item 加字段，这个断言自动覆盖新字段。

上面代码有一处故意的瑕疵留给你抓：第一套路由 r 建完之后从没被用——循环里的 r2 才是干活的（按角色参数化）。收拾干净（删掉 r，把 r2 的构造提为小函数）再提交，go vet 与你的眼睛都不该放过死变量——测试代码也是代码，收工时工作区不许留死代码。
