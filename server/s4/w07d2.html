验证码登录：sms-code 与 login

今天上两条接口：POST /api/homeos/auth/sms-code（发验证码）与 POST /api/homeos/auth/login（登录）。登录支持两种凭证：{phone, password} 走昨天的 bcrypt 校验，{phone, code} 走验证码校验。验证码是本地 mock——真实短信通道属课程范围外，但 store 的形态（有效期、限流、一次性消费）照真做，日后换通道只换一个「发送函数」。

今日目标

知识讲义

1. 验证码 store 的四个性质

性质规则不做会怎样
有效期发出后 5 分钟内可校验，过期作废 → 3003验证码永久有效 = 密码退化成 6 位数字
限流同一手机号 60 秒内只发一条；超限直接拒绝（教学期从简，也返回 3003 语义族的 400）被人拿你的接口轰炸别人的手机，短信费你出
一次性校验成功立刻删除；失败次数超过 5 次也删除同一验证码可重放，登录/注册/重置全被穿透
随机性crypto/rand 生成，不用 math/rand伪随机序列可被预测——这是安全边界，不是抽奖

教学期 store 用「map + sync.Mutex」的内存实现（Mutex 今天照抄，S5 并发阶段正式讲为什么必须加锁）。生产形态是带 TTL 的缓存或数据库表——形态可换，四个性质不变，所以值得先写对。mock 的是「发送」这一步：验证码不打真短信，而是写进服务日志，你从 make run 的终端里抄出来用。

2. 登录的防枚举：1002 一把兜底

昨天说好的重头戏。登录失败的三种真相——手机号没注册、密码错、验证码错——响应完全同形：

片段 · 三种失败一个样HTTP/1.1 401 Unauthorized
{"code":1002,"message":"手机号或凭证错误","data":null}

为什么验证码错也归 1002 而不归 3003？因为登录场景里验证码就是凭证本身，区分「号不存在」和「码不对」等于开放枚举。3003 留给注册与重置密码这类「手机号已经摆在明面上」的场景。service 层的写法是一条铁律：查无此人时，不提前 return，照样走一遍密码校验再统一失败——否则「号不存在」的响应比「密码错」快几十毫秒（bcrypt 的耗时差），计时侧信道照样能枚举。

3. 登录响应：token 字段今天先是占位

登录成功返回统一形状 data:{user, member, family, access_token, refresh_token}。member/family 是「当前会话家庭」的成员与家庭信息——刚注册还没建家庭时两者为 null，客户端接着走建家庭/受邀流程。token 两个字段今天先返回占位串 "dev-token-d3"：明天 D3 建 jwtutil 后换真签发（d1-d5 允许 handler 先用假数据，D6 前必须换真——本周五之前就会兑现）。

今日增量（在学习仓里落盘）

今天属增量类型 ②+⑤：加两条路由与 handler + 把 register 的验证码占位换成真 store。

步骤 1 · CodeStore

完整 · server/internal/homeos/service/smscode.gopackage service

import (
    "crypto/rand"
    "fmt"
    "log"
    "math/big"
    "sync"
    "time"
)

// CodeStore 验证码存储：内存 mock。真实短信通道属课程范围外，
// 日后接通道只需把 send 换成真实现，四个性质（有效期/限流/一次性/随机）不变。
type CodeStore struct {
    mu   sync.Mutex // S5 正式讲；今天记住「多请求并发读写 map 必须锁」
    codes map[string]codeEntry
}

type codeEntry struct {
    code     string
    expireAt time.Time
    sentAt   time.Time
    fails    int
}

func NewCodeStore() *CodeStore {
    return &CodeStore{codes: make(map[string]codeEntry)}
}

// Send 生成并「发送」验证码。60 秒限流；返回 false 表示被限流拒绝。
func (s *CodeStore) Send(phone string) bool {
    s.mu.Lock()
    defer s.mu.Unlock()
    now := time.Now()
    if e, ok := s.codes[phone]; ok && now.Sub(e.sentAt) < 60*time.Second {
        return false
    }
    n, err := rand.Int(rand.Reader, big.NewInt(1000000)) // crypto/rand，不是 math/rand
    if err != nil {
        return false
    }
    code := fmt.Sprintf("%06d", n.Int64())
    s.codes[phone] = codeEntry{code: code, expireAt: now.Add(5 * time.Minute), sentAt: now}
    log.Println("[sms-mock]", phone, "验证码:", code) // mock 发送 = 写日志
    return true
}

// Verify 校验并消费：成功即删除；过期/错误返回 false；连错 5 次作废。
func (s *CodeStore) Verify(phone, code string) bool {
    s.mu.Lock()
    defer s.mu.Unlock()
    e, ok := s.codes[phone]
    if !ok || time.Now().After(e.expireAt) {
        delete(s.codes, phone)
        return false
    }
    if e.code == code {
        delete(s.codes, phone) // 一次性：用过即毁
        return true
    }
    e.fails++
    if e.fails >= 5 {
        delete(s.codes, phone)
    } else {
        s.codes[phone] = e
    }
    return false
}

步骤 2 · service：SendCode 与 Login

片段 · server/internal/homeos/service/auth.go 追加（前置：昨天的 AuthService 与 passutil）// AuthService 增加一个字段 codes *CodeStore，NewAuthService 相应加参。

// SendCode 发验证码。scene 仅作日志标记（login/register/reset），store 不区分。
func (s *AuthService) SendCode(phone, scene string) error {
    if !phoneRE.MatchString(phone) {
        return response.ErrInvalidParam // → 1004 / 400
    }
    if !s.codes.Send(phone) {
        return response.ErrRateLimited // → 400，message「发送太频繁，请稍后再试」
    }
    return nil
}

// LoginResult 登录响应 data 的形状。token 今天占位，S4·D3 换真。
type LoginResult struct {
    User         *model.User   `json:"user"`
    Member       *model.Member `json:"member"`  // 无家庭时为 null
    Family       *model.Family `json:"family"`  // 无家庭时为 null
    AccessToken  string        `json:"access_token"`
    RefreshToken string        `json:"refresh_token"`
}

// Login 双通道：password 非空走密码，code 非空走验证码。二者必须恰好给一个。
func (s *AuthService) Login(ctx context.Context, phone, password, code string) (*LoginResult, error) {
    if (password == "") == (code == "") {
        return nil, response.ErrInvalidParam // 给 0 个或 2 个凭证都是参数错 → 1004
    }
    u, err := s.users.FindByPhone(ctx, phone)
    if err != nil && !errors.Is(err, response.ErrNotFound) {
        return nil, fmt.Errorf("login: %w", err)
    }
    if u == nil {
        // 防枚举 + 防计时侧信道：查无此人也「演」一遍校验，再统一失败
        if password != "" {
            passutil.CheckPassword(dummyHash, password) // dummyHash 是包级预生成的合法哈希
        } else {
            s.codes.Verify(phone, code)
        }
        return nil, response.ErrAuthFailed // → 1002 / 401，message 与凭证错完全一致
    }
    ok := false
    if password != "" {
        ok = passutil.CheckPassword(u.PasswordHash, password)
    } else {
        ok = s.codes.Verify(phone, code)
    }
    if !ok {
        return nil, response.ErrAuthFailed // 同一个 1002，不区分密码错还是码错
    }
    // 当前会话家庭 = 用户的第一个家庭（S4·D9 起可切换）；没有则为 null
    member, family := s.primarySession(ctx, u.ID)
    return &LoginResult{
        User: u, Member: member, Family: family,
        AccessToken: "dev-token-d3", RefreshToken: "dev-token-d3", // TODO(S4·D3)：jwtutil 真签发
    }, nil
}

哨兵错误集今天再补两个（与 CodeOf 映射对齐）：ErrAuthFailed（→ 1002 / 401）、ErrRateLimited（→ 400，message 提示稍后再试；不新造业务码）。primarySession 今天可以先返回 (nil, nil)——本周还没有家庭数据，S4·D7 建家庭后回来补真查询（又是一次类型⑤）。

步骤 3 · handler 与路由

片段 · server/internal/homeos/handler/auth.go 追加（前置：昨天的 AuthHandler）type smsCodeReq struct {
    Phone string `json:"phone" binding:"required"`
    Scene string `json:"scene" binding:"required,oneof=login register reset"`
}

type loginReq struct {
    Phone    string `json:"phone" binding:"required"`
    Password string `json:"password"`
    Code     string `json:"code"`
}

// SendCode 处理 POST /api/homeos/auth/sms-code
func (h *AuthHandler) SendCode(c *gin.Context) {
    var req smsCodeReq
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, 400, response.CodeInvalidParam, "参数错误")
        return
    }
    if err := h.auth.SendCode(req.Phone, req.Scene); err != nil {
        response.FailErr(c, err)
        return
    }
    response.OK(c, gin.H{"sent": true})
}

// Login 处理 POST /api/homeos/auth/login
func (h *AuthHandler) Login(c *gin.Context) {
    var req loginReq
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, 400, response.CodeInvalidParam, "参数错误")
        return
    }
    res, err := h.auth.Login(c.Request.Context(), req.Phone, req.Password, req.Code)
    if err != nil {
        response.FailErr(c, err)
        return
    }
    response.OK(c, res)
}

路由组加两行：auth.POST("/sms-code", ...)、auth.POST("/login", ...)。然后回 register 的 service，把 code != "123456" 的占位换成 s.codes.Verify(phone, code)，校验失败返回 ErrCaptchaInvalid（注册场景手机号是新的，不存在枚举问题，3003 可以放心用）。昨天测试里的 mock 验证码用例同步改：先 Send 再拿真码注册。

import (
    "context"
    "errors"
    "testing"

    "homecube/internal/homeos/repo"
    "homecube/pkg/response"
)

func TestLoginEnumeration(t *testing.T) {
    users := repo.NewMemoryUserRepo()
    codes := NewCodeStore()
    svc := NewAuthService(users, codes)
    ctx := context.Background()

    // 注册一个真用户（验证码走真 store）
    codes.Send("13800000001")
    if _, err := svc.Register(ctx, "13800000001", lastCode(t, codes, "13800000001"), "secret99"); err != nil {
        t.Fatalf("注册失败: %v", err)
    }

    _, err1 := svc.Login(ctx, "13800000001", "wrongpass1", "")   // 号在、密码错
    _, err2 := svc.Login(ctx, "13999999999", "whatever1", "")    // 号不存在
    if !errors.Is(err1, response.ErrAuthFailed) || !errors.Is(err2, response.ErrAuthFailed) {
        t.Fatal("两种失败必须同为 1002（ErrAuthFailed）")
    }
    if err1.Error() != err2.Error() {
        t.Fatalf("响应 message 必须同形: %q vs %q", err1, err2)
    }

    // 验证码通道：发码 → 码登录成功；旧码重放 → 1002
    codes.Send("13800000001")
    code := lastCode(t, codes, "13800000001")
    if _, err := svc.Login(ctx, "13800000001", "", code); err != nil {
        t.Fatalf("验证码登录应成功: %v", err)
    }
    if _, err := svc.Login(ctx, "13800000001", "", code); !errors.Is(err, response.ErrAuthFailed) {
        t.Fatal("验证码重放必须失败（一次性消费）")
    }
}

常见坑（新手 100% 会踩）

坑现象正确姿势
用 math/rand 生成验证码默认种子可预测，理论上能推算下一个码安全边界一律 crypto/rand
验证码校验成功后不删除同一个码反复登录，成了永久后门一次性消费：Verify 成功即 delete
「号不存在」提前 return响应时间差暴露注册名单（计时侧信道）查无此人也走一遍 bcrypt 假校验再统一 1002
登录区分「密码错」和「验证码错」的 messagemessage 不同 = 枚举通道又开了登录场景凭证错一律同一 message
map 不加锁并发读写Go 运行时直接 fatal error: concurrent map writes，进程崩Mutex 护住所有读写；S5 会讲还有没有别的选择
限流只做在前端（按钮置灰 60 秒）curl 绕过前端无限发限流必须在 store/服务端这一层
把占位 token 忘了换D3 之后登录还返回 dev-token-d3TODO 注释里写明兑现日（S4·D3）；D6 收口前必须清零——留着就是「注册了但没实现」的假入口
