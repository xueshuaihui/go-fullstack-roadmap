周六 · 忘记/重置密码，登录全链路测试

本周收尾。上午两条新接口：POST /api/homeos/auth/forgot-password 与 POST /api/homeos/auth/reset-password——认证 7 条到此全部落地。下午把 D1-D5 的五个微增量串成一条全链路测试：注册 → 发码 → 登录 → 刷新轮换 → 登出 → 忘记 → 重置 → 新密码重登 → 旧 refresh 作废。这条测试以后每次改动都要跑，它是你认证域的回归防线。

今日目标

知识讲义

1. 忘记密码：一场「不能说出真相」的接口设计

忘记流程分两步：forgot 发重置验证码，reset 带码和新密码来换。难点全在 forgot 的防枚举上——它比登录更凶险：登录还要先知道密码格式，forgot 只要一个手机号就能问。规则只有一条：无论手机号是否注册，响应完全一致：

片段 · 两种真相一个响应HTTP/1.1 200 OK
{"code":0,"message":"ok","data":{"sent":true}}   // 号在：真发码。号不在：也回这个，什么都不做

注意连状态码和业务码都不给差异——「号不存在」返回 1001 或任何 distinguishing 信息都等于开放注册名单查询。限流照旧（60 秒一条，走 D2 的 CodeStore），限流拒绝的响应同样不透露号码存在性。

2. 重置成功 = 全设备下线

用户为什么要重置密码？大概率是「怀疑密码泄露」。所以 reset 成功的瞬间，必须把该用户所有 refresh token 撤销（RevokeAllByUser）——偷密码的人即使已经登录，7 天的续命通道也被一刀切断。access token 呢？它无状态、撤不回，但最多再活 15 分钟，且中间件还压着成员状态与 pver 两道校验。这是「短 TTL + 有状态 refresh」组合拳第一次完整亮相：access 管性能，refresh 管撤销，TTL 管兜底。

reset 的错误出口分三档，别混：验证码错/过期 → 3003（400，手机号已在明面，无枚举可防）；新密码不合策略 → 1004（400）；手机号压根没注册还带着「正确格式」的码来重置 → 验证码在 store 里对不上号，仍走 3003。全程不出现 1001/1002——这两个码属于「登录与查询」语境。

3. 全链路测试怎么组织

认证域的九个动作有严格的先后依赖，适合写成一个长测试函数按序断言（不是表驱动——步骤间有状态传递）。装配用 httptest：内存 repo 全家桶 + 随机测试密钥 + 真路由，进程内直调，毫秒级跑完。每步只断言两件事：HTTP 状态 + body 里的 code。这条测试的另一个身份是活文档：新读者跑一遍、读一遍，就懂你的认证域全貌。

步骤 1 · service：Forgot 与 Reset

片段 · server/internal/homeos/service/auth.go 追加（前置：本周的 AuthService / CodeStore / hashToken）// ForgotPassword 发重置验证码。防枚举：号在不在，响应与耗时形态都一致。
func (s *AuthService) ForgotPassword(ctx context.Context, phone string) error {
    if !phoneRE.MatchString(phone) {
        return response.ErrInvalidParam // 格式都不对 → 1004，这不算泄露（与具体号码无关）
    }
    u, err := s.users.FindByPhone(ctx, phone)
    if err != nil && !errors.Is(err, response.ErrNotFound) {
        return fmt.Errorf("forgot: %w", err) // 系统错 → 5000
    }
    if u == nil {
        return nil // 号不存在：静默成功。不发码、不报错、不留差异
    }
    if !s.codes.Send(phone) {
        return response.ErrRateLimited // 限流与号码存在性无关，两种号码同形，可安全返回
    }
    return nil
}

// ResetPassword 验码 → 换哈希 → 撤销该用户全部 refresh。
func (s *AuthService) ResetPassword(ctx context.Context, phone, code, newPassword string) error {
    if !passutil.ValidPassword(newPassword) {
        return response.ErrInvalidParam // → 1004
    }
    if !s.codes.Verify(phone, code) {
        return response.ErrCaptchaInvalid // → 3003：码错/过期/号不存在，同一出口
    }
    u, err := s.users.FindByPhone(ctx, phone)
    if err != nil || u == nil {
        // 码都对上了号却不在？store 与库不一致，属系统错，不冒充 3003
        return fmt.Errorf("reset: user vanished: %w", response.ErrInternal)
    }
    hash, err := passutil.HashPassword(newPassword)
    if err != nil {
        return fmt.Errorf("reset: %w", err)
    }
    if err := s.users.UpdatePasswordHash(ctx, u.ID, hash); err != nil {
        return fmt.Errorf("reset: update: %w", err)
    }
    if err := s.refreshTokens.RevokeAllByUser(ctx, u.ID); err != nil { // 全设备下线
        return fmt.Errorf("reset: revoke all: %w", err)
    }
    return nil
}

上面的 ForgotPassword 有一处要你动手想：限流分支返回什么才「对两种号码同形」？答案写在注释里了——限流与号码存在性无关，两种号码都会撞限流，所以返回 ErrRateLimited 是安全的；把那一行 return nil 改成 return response.ErrRateLimited 落盘。RefreshTokenRepo 接口今天补齐第四个方法 RevokeAllByUser（GORM 一条 UPDATE ... WHERE user_id=? AND revoked_at IS NULL）。

步骤 2 · handler 与路由

片段 · server/internal/homeos/handler/auth.go 追加（前置：本周的 AuthHandler）type forgotReq struct {
    Phone string `json:"phone" binding:"required"`
}

type resetReq struct {
    Phone       string `json:"phone" binding:"required"`
    Code        string `json:"code" binding:"required"`
    NewPassword string `json:"new_password" binding:"required"`
}

// ForgotPassword 处理 POST /api/homeos/auth/forgot-password
func (h *AuthHandler) ForgotPassword(c *gin.Context) {
    var req forgotReq
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, 400, response.CodeInvalidParam, "参数错误")
        return
    }
    if err := h.auth.ForgotPassword(c.Request().Context(), req.Phone); err != nil {
        response.FailErr(c, err)
        return
    }
    response.OK(c, gin.H{"sent": true}) // 号在号不在，都是这一句
}

// ResetPassword 处理 POST /api/homeos/auth/reset-password
func (h *AuthHandler) ResetPassword(c *gin.Context) {
    var req resetReq
    if err := c.ShouldBindJSON(&req); err != nil {
        response.Fail(c, 400, response.CodeInvalidParam, "参数错误")
        return
    }
    if err := h.auth.ResetPassword(c.Request().Context(), req.Phone, req.Code, req.NewPassword); err != nil {
        response.FailErr(c, err)
        return
    }
    response.OK(c, gin.H{"reset": true})
}

import (
    "encoding/json"

    "strings"
    "testing"

    "github.com/gin-gonic/gin"

// do 发一个 JSON 请求，返回 HTTP 状态与统一响应体的 code。
func do(t *testing.T, r *gin.Engine, method, path, body string) (int, int, string) {
    t.Helper()
    w := httptest.NewRecorder()
    req := httptest.NewRequest(method, path, strings.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    r.ServeHTTP(w, req)
    var env struct {
        Code int    `json:"code"`
        Data map[string]any `json:"data"`
    }
    if err := json.Unmarshal(w.Body.Bytes(), &env); err != nil {
        t.Fatalf("响应不是统一信封: %s", w.Body.String())
    }
    raw, _ := env.Data["refresh_token"].(string)
    return w.Code, env.Code, raw
}

抄前必读：骨架只给你接好了前两步，第 3-9 步亲手接——串测试的价值就在接线本身。每步都形如上面 do(...) 的三行断言，逐步要求：登录断言拿到 access/refresh（data 里两个 token 非空）；refresh 断言轮换成功且旧票重放 c == 1005；logout 后再 refresh 断言 1005；forgot 对注册过的号断言 c == 0、对 13999999999 也断言 c == 0（防枚举对照）；reset 用错码断言 3003、用 peek() 真码断言成功；旧密码登录断言 1002、新密码登录断言成功；最后拿 reset 之前签发的 refresh 再刷新，断言 1005（全设备下线生效）。

常见坑（新手 100% 会踩）

坑现象正确姿势
forgot 对未注册号返回 1001接口变成注册名单探测器号在号不在都是 200 + {"sent":true}
reset 只撤当前设备的 refresh用户以为改密码就全下线了，其他设备的会话还活着RevokeAllByUser：重置 = 全设备下线
reset 成功复用旧验证码Verify 没消费，码还能再重置一次验证码一次性（D2 的性质三），reset 成功前 Verify 已删码
新密码与旧密码相同不拦「重置」了个寂寞教学期可不拦（清单没要求）；要拦就在 service 加一条 CheckPassword(old) 判断，别写进 handler
全链路测试步骤间共享错变量第三步用了第一步的旧 refresh，断言假绿每步的产物立刻赋给命名清晰的变量（access1/refresh1/refresh2…）
测试里用生产 .env 的密钥测试机密钥泄露进断言输出测试密钥用随机生成或固定 test-secret，与 .env 无关
把 3003 用到登录接口登录的验证码错回了 3003，枚举通道重开登录语境一律 1002；3003 只属于注册/重置
